     1→// Prisma schema for the OTP Email Verification Service with production-grade
     2→// security layer (device fingerprinting, IP blocking, disposable-email blocklist,
     3→// VPN/proxy policy, brute-force & account lockout, security event audit).

     7→// your Neon pooled connection string. IDs use `Int @default(autoincrement())`
     8→// (works on SQLite + PostgreSQL); `Bytes` (codeHash) is supported by both.

   462→// Reusable transactional email templates that future messaging features will

   464→// not OTP-specific. OTP behavior and EmailTheme are unchanged by this phase.
   465→//
   466→// Parent template holds mutable metadata (name, slug, description).
   467→// `currentVersion` points at the latest immutable content snapshot.
   468→// Slug is tenant-unique: (userId, slug). Same slug may exist for two users.

   488→// Immutable content snapshot of a template. NEVER UPDATED after creation.
   489→// Each content edit inserts a new version row with version = parent.currentVersion
   490→// (atomically incremented inside a transaction). Historical versions cannot be
   491→// edited or deleted individually — only the parent template can be deleted,
   492→// which cascades all versions.
   493→//
   494→// `variables` is a deterministic, de-duplicated, sorted list of flat variable
   495→// names (e.g. ["name", "order_id"]) extracted from subject+html+text.
   496→model TransactionalTemplateVersion {
   497→  id         Int      @id @default(autoincrement())
   498→  templateId Int
   499→  version    Int
   500→  subject    String
   501→  html       String
   502→  text       String?
   503→  variables  Json
   504→  createdAt  DateTime @default(now())
   505→
   506→  template TransactionalTemplate @relation(fields: [templateId], references: [id], onDelete: Cascade)
   507→
   508→  @@unique([templateId, version])
   509→  @@index([templateId, createdAt])

   512→// ---- Email Messages (Phase 4) ---------------------------------------------
   513→// Persisted record of every transactional email send attempt. Stores audit
   514→// metadata (recipient, rendered subject, status, provider ids) but NOT the
   515→// full rendered HTML body or variable values — those may contain sensitive
   516→// application data the caller does not want persisted indefinitely.
   517→//
   518→// Idempotency: (userId, idempotencyKeyHash) is unique. The idempotency key is
   519→// SHA-256 hashed before storage (never plaintext). A separate requestFingerprint
   520→// (SHA-256 of the canonicalized request body) lets us detect same-key-different-body
   521→// conflicts and return 409.
   522→//
   523→// Status lifecycle: pending -> sent | failed | rejected.
   524→//   - pending: claim created, delivery in progress
   525→//   - sent: provider accepted and returned a message id
   526→//   - failed: provider call errored (safe error classification persisted)
   527→//   - rejected: request claimed but denied before delivery (e.g. quota exhaustion)
   528→model EmailMessage {
   529→  id                  Int       @id @default(autoincrement())
   530→  messageId           String    @unique @default(uuid())
   531→  userId              Int
   532→  templateId          Int?
   533→  templateVersion     Int?
   534→  toEmail             String
   535→  subject             String
   536→  status              String    // "pending" | "sent" | "failed" | "rejected"
   537→  provider            String?
   538→  providerMessageId   String?
   539→  source              String    // "api_v1" | "dashboard_test"
   540→  environment         String?
   541→  requestId           String?
   542→  idempotencyKeyHash  String?
   543→  requestFingerprint  String?
   544→  errorCode           String?
   545→  errorMessage        String?
   546→  createdAt           DateTime  @default(now())
   547→  sentAt              DateTime?
   548→  failedAt            DateTime?
   549→
   550→  user     User                  @relation(fields: [userId], references: [id])
   551→  template TransactionalTemplate? @relation(fields: [templateId], references: [id], onDelete: SetNull)
   552→
   553→  @@unique([userId, idempotencyKeyHash])
   554→  @@index([userId, createdAt])
   555→  @@index([userId, status, createdAt])
   556→  @@index([toEmail, createdAt])

   765→// ---- Consent History (Phase 9) ---------------------------------------------
   766→// Immutable, tenant-owned audit trail of every marketing-consent transition
   767→// for a Contact. One row per ACTUAL state transition. Never updated — only
   768→// appended. Same-state repeated requests do NOT create fake transition rows
   769→// (no-op semantics — see service).

   771→// Phase 9 canonical marketing states (preserved from Phase 1 — no destructive
   772→// rewrite): "unknown" | "subscribed" | "unsubscribed".

   774→// `source` is one of: "dashboard" | "api" | "unsubscribe" | "import" | "system".
   775→// Phase 9 only writes "dashboard", "api", "unsubscribe", and "system".
   776→// "import" is reserved for future Phase 11 use (import-time explicit consent
   777→// capture) — Phase 9 imports MUST NOT create consent events.

   900→// ---- Consent Mutation Idempotency (Phase 9) --------------------------------
   901→// Durable idempotency outcome store for consent/suppression mutations.
   902→// Decoupled from transition-event tables so that a NO-OP request (e.g.
   903→// subscribing an already-subscribed contact) can be durably replayed without
   904→// creating a fake transition event.
   905→//
   906→// A no-op request with an idempotency key stores resultStatus="no_op" here.
   907→// A retry with the same key (even after the contact's state has changed)
   908→// replays the original no_op outcome — it does NOT re-evaluate and potentially
   909→// apply a transition that the original request did not apply.
   910→//
   911→// operation is the canonical action name ("subscribe" | "unsubscribe" |
   912→// "suppress" | "unsuppress"). The unique constraint on
   913→// (userId, operation, idempotencyKeyHash) is the idempotency namespace —
   914→// the same key reused for a different operation does NOT replay.
   915→//
   916→// targetType + targetKey encode the mutation target:
   917→//   - "contact" -> targetKey = String(contactId)
   918→//   - "email"   -> targetKey = normalizedEmail
   919→// Verified before replay to prevent cross-target key reuse.
   920→//
   921→// requestFingerprint is the SHA-256 of the canonicalized mutable payload.
   922→// A retry with the same key but a different fingerprint throws
   923→// IdempotencyConflictError (409).
   924→model ConsentMutationIdempotency {
   925→  id                   Int      @id @default(autoincrement())
   926→  userId               Int
   927→  operation            String   // "subscribe" | "unsubscribe" | "suppress" | "unsuppress"
   928→  targetType           String   // "contact" | "email"
   929→  targetKey            String   // String(contactId) or normalizedEmail
   930→  idempotencyKeyHash   String
   931→  requestFingerprint   String?
   932→  resultStatus         String   // "applied" | "no_op" | "not_suppressed"
   933→  resultEventId        String?  // public UUID of the transition event (null for no_op)
   934→  resultSuppressionId  String?  // public UUID of the suppression entry (for suppress ops)
   935→  createdAt            DateTime @default(now())
   936→
   937→  @@unique([userId, operation, idempotencyKeyHash])
   938→  @@index([userId, operation, idempotencyKeyHash])
   939→  @@index([userId, targetType, targetKey, createdAt])

   999→// ---- Broadcast Recipient (Phase 10) ----------------------------------------
  1000→// Per-recipient send record. One row per (broadcastId, contactId). Created at
  1001→// launch time via INSERT...SELECT (DB-side snapshot — no application memory
  1002→// materialization for large audiences).
  1003→//
  1004→// STRUCTURAL TENANT ISOLATION: composite FK
  1005→//   (userId, broadcastId) → Broadcast(userId, id)
  1006→// The DB itself rejects cross-tenant recipient references to the parent broadcast.
  1007→// The contactId FK is single-column (ON DELETE SET NULL) — see "CONTACT DELETION"
  1008→// below for the rationale. Tenant agreement on contactId is enforced at the
  1009→// application layer by reading both rows through the recipient's userId (which
  1010→// is NOT NULL and structurally tied to the parent broadcast).
  1011→//
  1012→// COUNTS ARE DERIVED from these rows (total/pending/processing/dispatching/
  1013→// sent/skipped/failed). No retry-sensitive increment counters as source of truth.
  1014→//
  1015→// RECIPIENT STATE MACHINE (with durable exclusive dispatch):
  1016→//   pending → processing → dispatching → sent | skipped | failed
  1017→//
  1018→//   - pending:      queued for any worker to claim.
  1019→//   - processing:   claimed by a worker via CAS (lockedBy=workerId). NO external
  1020→//                   I/O has occurred yet. Safe to stale-recover → pending.
  1021→//   - dispatching:  durable exclusive dispatch state established via
  1022→//                   CAS (processing → dispatching WHERE lockedBy=workerId).
  1023→//                   ONLY this worker may call provider.send(). NEVER
  1024→//                   auto-requeued — stale dispatching rows become terminal
  1025→//                   `failed` with errorCode=provider_outcome_unknown after
  1026→//                   a safe timeout (recoverAbandonedDispatches).
  1027→//   - sent/skipped/failed: terminal. Terminal CAS verifies lockedBy=workerId.
  1028→//
  1029→// CONTACT DELETION: `contactId` is nullable + single-column FK with
  1030→// ON DELETE SET NULL. Deleting a contact preserves the recipient row's audit
  1031→// history (sentAt, errorCode remain intact) by nullifying only contactId.
  1032→// The `userId` column remains NOT NULL — single-column SET NULL is safe because
  1033→// only `contactId` is nullified. At processRecipient time, a null contactId is
  1034→// skipped with skipReason=contact_not_found.
  1035→// We use a single-column FK (rather than the composite (userId, contactId) FK)
  1036→// because PostgreSQL's composite `ON DELETE SET NULL` would attempt to null ALL
  1037→// FK columns including the NOT NULL `userId`, which would fail. Single-column
  1038→// SET NULL on `contactId` alone is the safe form. See the Phase 9 lesson in
  1039→// agent-lessons.md about composite SET NULL with NOT NULL tenant columns.
  1040→model BroadcastRecipient {
  1041→  id                  Int       @id @default(autoincrement())
  1042→  userId              Int
  1043→  broadcastId         Int
  1044→  contactOwnerUserId  Int?
  1045→  contactId           Int?
  1046→  status              String    @default("pending") // pending|processing|dispatching|sent|skipped|failed
  1047→  skipReason          String?   // not_subscribed|suppressed|contact_not_found|broadcast_cancelled|quota_exhausted
  1048→  lockedAt           DateTime?

  1068→// ---- Broadcast Mutation Idempotency (Phase 10) -----------------------------
  1069→// Durable idempotency outcome store for stateful broadcast mutations
  1070→// (launch, cancel). Decoupled from Broadcast row state so a no-op request
  1071→// (e.g. cancelling an already-cancelled broadcast) can be durably replayed
  1072→// without re-evaluating against changed state.
  1073→//
  1074→// Identity: (userId, operation, idempotencyKeyHash) — binds the key to both
  1075→// the tenant and the operation. Same key + different operation → not a replay.
  1076→//
  1077→// Conflict detection: `requestFingerprint` (SHA-256 of canonicalized body)
  1078→// detects same-key-different-payload conflicts → 409 idempotency_conflict.
  1079→//
  1080→// Target verification: `targetBroadcastId` is verified before replay to
  1081→// prevent cross-target key reuse.
  1082→model BroadcastMutationIdempotency {
  1083→  id                  Int      @id @default(autoincrement())
  1084→  userId              Int
  1085→  operation           String   // "launch" | "cancel"
  1086→  targetBroadcastId   String   // public broadcastId UUID
  1087→  idempotencyKeyHash  String
  1088→  requestFingerprint  String?
  1089→  resultStatus        String   // "applied" | "no_op" | "idempotent_replay"
  1090→  resultData          Json?
  1091→  createdAt           DateTime @default(now())
  1092→
  1093→  user User @relation(fields: [userId], references: [id], onDelete: Cascade)
  1094→
  1095→  @@unique([userId, operation, idempotencyKeyHash])
  1096→  @@index([userId, operation, idempotencyKeyHash])
  1097→  @@index([userId, targetBroadcastId])
  1098→}
