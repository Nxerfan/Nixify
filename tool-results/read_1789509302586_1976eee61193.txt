     1→/**
     2→ * Phase 10 — Broadcast service.
     3→ *
     4→ * Central service for all broadcast lifecycle operations:
     5→ *   - Draft CRUD (create, get, list, update, delete)
     6→ *   - Audience preview (DB-side aggregation — no N+1, no unbounded lists)
     7→ *   - Launch (DB-side INSERT...SELECT audience snapshot + freeze content +
     8→ *     review threshold check + Idempotency-Key dedup)
     9→ *   - Cancel (idempotent + Idempotency-Key dedup)
    10→ *   - Recipient claiming (atomic CAS, stale recovery — processing only)
    11→ *   - Dispatch state (processing → dispatching CAS before provider call;
    12→ *     stale dispatching → terminal failed, NEVER auto-requeued)
    13→ *   - Send processing (eligibility re-check + render + provider dispatch)
    14→ *   - Finalization (0 pending + 0 processing + 0 dispatching → completed)
    15→ *
    16→ * INVARIANTS:
    17→ *   1. Audience membership is snapshotted at launch via DB-side INSERT...SELECT.
    18→ *      No full-audience ID array is materialized in Node memory.
    19→ *   2. Consent/suppression is NOT snapshotted — it's re-checked immediately
    20→ *      before provider dispatch via `getMarketingEligibility()`.
    21→ *   3. Counts are DERIVED from BroadcastRecipient rows (no retry-sensitive counters).
    22→ *   4. Content is frozen at launch — no editing after draft status.
    23→ *   5. Structural tenant isolation via composite FK on (userId, broadcastId).
    24→ *   6. BROADCAST_EMAILS quota consumed exactly once per actual provider attempt.
    25→ *      Skipped recipients consume no quota. Idempotent replays consume no second quota.
    26→ *   7. BROADCAST_EMAILS access is enforced centrally in this service (in addition
    27→ *      to CONTACTS at the route boundary). Null-owner/system v1 keys cannot act
    28→ *      as Broadcast tenants.
    29→ *   8. The `dispatching` state is the durable exclusive dispatch marker — only
    30→ *      the worker that wins the `processing → dispatching` CAS may call
    31→ *      `provider.send()`. Stale dispatching rows become terminal `failed` with
    32→ *      errorCode=provider_outcome_unknown; they are NEVER auto-requeued.
    33→ *   9. Contact deletion → recipient row survives (contactId nullable + ON DELETE
    34→ *      SET NULL). processRecipient handles contactId === null → skipped with
    35→ *      contact_not_found.
    36→ */

   173→// ---- Idempotency helpers --------------------------------------------------
   174→
   175→function hashIdempotencyKey(userId: number, operation: string, key: string): string {
   176→  return createHash("sha256").update(`${userId}:${operation}:${key}`).digest("hex");
   177→}
   178→
   179→function hashRequestFingerprint(payload: unknown): string {
   180→  // Canonical JSON: stable property order via JSON.stringify (deterministic
   181→  // for primitive key order in V8). Sufficient for detecting same-key-different-body.
   182→  const json = JSON.stringify(payload ?? null);
   183→  return createHash("sha256").update(json).digest("hex");
   184→}
   185→
   186→function validateIdempotencyKey(key: string | undefined): string | null {
   187→  if (!key) return null;
   188→  if (key.length < IDEMPOTENCY_KEY_MIN || key.length > IDEMPOTENCY_KEY_MAX) {
   189→    throw new BroadcastValidationError(`Idempotency-Key must be ${IDEMPOTENCY_KEY_MIN}-${IDEMPOTENCY_KEY_MAX} chars.`);
   190→  }
   191→  return key;
   192→}
   193→
   194→/**
   195→ * Derive a 64-bit lock key for idempotency serialization.
   196→ * Packs: high 32 bits = userId, low 32 bits = hash(operation + keyHash).
   197→ * Same (userId, operation, keyHash) → same lock key → serialized.
   198→ * Different tenants/operations/keys → independent locks.
   199→ */
   200→function canonicalIdempotencyLockKey(userId: number, operation: string, keyHash: string): bigint {
   201→  const tenant = BigInt(userId) & BigInt("0xffffffff");
   202→  const opHashHex = createHash("sha256").update(`${operation}:${keyHash}`).digest("hex").slice(0, 8);
   203→  const opHash = BigInt(parseInt(opHashHex, 16)) & BigInt("0xffffffff");
   204→  return (tenant << BigInt(32)) | opHash;
   205→}

   209→export async function createBroadcast(input: CreateBroadcastInput): Promise<BroadcastSummary> {
   210→  const { userId, name, subject, htmlContent, textContent, audienceType, targetGroupId } = input;
   211→
   212→  // Central BROADCAST_EMAILS capability gate (in addition to CONTACTS at route).
   213→  await requireBroadcastAccess(userId);
   214→  if (audienceType === AUDIENCE_TYPES.GROUP) {
   215→    await requireGroupsAccess(userId);
   216→  }
   217→
   218→  // Full draft-content validation (used by create AND update — same rules).
   219→  const validation = await validateFullDraft(
   220→    { name, subject, htmlContent, textContent, audienceType, targetGroupId },
   221→    {
   222→      userId,
   223→      verifyGroupOwnership: async (gid, uid) => {
   224→        const g = await db.group.findFirst({ where: { id: gid, userId: uid }, select: { id: true } });
   225→        return !!g;
   226→      },
   227→    },
   228→  );
   229→  if (!validation.valid) {
   230→    throw new BroadcastValidationError(validation.error!);
   231→  }
   232→
   233→  const broadcast = await db.broadcast.create({
   234→    data: {
   235→      userId,
   236→      name: name.trim(),
   237→      subject,
   238→      htmlContent,
   239→      textContent: textContent ?? null,
   240→      audienceType,
   241→      targetGroupId: targetGroupId ?? null,
   242→      status: BROADCAST_STATUSES.DRAFT,
   243→      reviewStatus: REVIEW_STATUSES.NOT_REQUIRED,
   244→    },
   245→  });
   246→
   247→  return toSummary(broadcast, { totalRecipients: 0, pendingCount: 0, processingCount: 0, dispatchingCount: 0, sentCount: 0, skippedCount: 0, failedCount: 0 });
   248→}

   260→export async function listBroadcasts(
   261→  userId: number,
   262→  opts: { page?: number; pageSize?: number; status?: string } = {},
   263→): Promise<{ broadcasts: BroadcastSummary[]; total: number; page: number; pageSize: number }> {
   264→  await requireBroadcastAccess(userId);
   265→  const page = Math.max(1, opts.page ?? 1);
   266→  const pageSize = Math.min(50, Math.max(1, opts.pageSize ?? 20));
   267→  const where: Record<string, unknown> = { userId };
   268→  if (opts.status) where.status = opts.status;
   269→
   270→  const [rows, total] = await Promise.all([
   271→    db.broadcast.findMany({
   272→      where,
   273→      orderBy: { createdAt: "desc" },
   274→      skip: (page - 1) * pageSize,
   275→      take: pageSize,
   276→    }),
   277→    db.broadcast.count({ where }),
   278→  ]);
   279→
   280→  const summaries: BroadcastSummary[] = [];
   281→  for (const row of rows) {
   282→    summaries.push(toSummary(row, await deriveCounts(row.id)));
   283→  }
   284→  return { broadcasts: summaries, total, page, pageSize };
   285→}

   287→export async function updateBroadcast(
   288→  userId: number,
   289→  broadcastId: string,
   290→  updates: { name?: string; subject?: string; htmlContent?: string; textContent?: string | null; audienceType?: AudienceType; targetGroupId?: number | null },
   291→): Promise<BroadcastSummary | null> {
   292→  await requireBroadcastAccess(userId);
   293→
   294→  const broadcast = await db.broadcast.findFirst({
   295→    where: { broadcastId, userId },
   296→    select: { id: true, status: true, name: true, subject: true, htmlContent: true, textContent: true, audienceType: true, targetGroupId: true },
   297→  });
   298→  if (!broadcast) return null;
   299→
   300→  if (!EDITABLE_STATUSES.has(broadcast.status)) {
   301→    throw new BroadcastValidationError("Broadcast content can only be edited while in draft status.");
   302→  }
   303→
   304→  // Build the MERGED draft state and validate the entire resulting state.
   305→  const merged: {
   306→    name: string;
   307→    subject: string;
   308→    htmlContent: string;
   309→    textContent: string | null;
   310→    audienceType: string;
   311→    targetGroupId: number | null;
   312→  } = {
   313→    name: updates.name !== undefined ? updates.name.trim() : broadcast.name,
   314→    subject: updates.subject !== undefined ? updates.subject : broadcast.subject,
   315→    htmlContent: updates.htmlContent !== undefined ? updates.htmlContent : broadcast.htmlContent,
   316→    textContent: updates.textContent !== undefined ? updates.textContent : broadcast.textContent,
   317→    audienceType: updates.audienceType !== undefined ? updates.audienceType : broadcast.audienceType,
   318→    targetGroupId:
   319→      updates.targetGroupId !== undefined
   320→        ? updates.targetGroupId ?? null
   321→        : broadcast.targetGroupId,
   322→  };
   323→
   324→  // If audienceType=group in the merged state, also require GROUPS access.
   325→  if (merged.audienceType === AUDIENCE_TYPES.GROUP) {
   326→    await requireGroupsAccess(userId);
   327→  }
   328→
   329→  const validation = await validateFullDraft(merged, {
   330→    userId,
   331→    verifyGroupOwnership: async (gid, uid) => {
   332→      const g = await db.group.findFirst({ where: { id: gid, userId: uid }, select: { id: true } });
   333→      return !!g;
   334→    },
   335→  });
   336→  if (!validation.valid) {
   337→    throw new BroadcastValidationError(validation.error!);
   338→  }
   339→
   340→  // If audienceType=all_contacts, force targetGroupId to null.
   341→  const data: Record<string, unknown> = {
   342→    name: merged.name,
   343→    subject: merged.subject,
   344→    htmlContent: merged.htmlContent,
   345→    textContent: merged.textContent,
   346→    audienceType: merged.audienceType,
   347→    targetGroupId: merged.audienceType === AUDIENCE_TYPES.ALL_CONTACTS ? null : merged.targetGroupId,
   348→  };
   349→
   350→  const updated = await db.broadcast.update({
   351→    where: { id: broadcast.id },
   352→    data,
   353→  });
   354→  return toSummary(updated, await deriveCounts(updated.id));
   355→}

   371→// ---- Preview (DB-side aggregation) ---------------------------------------
   372→
   373→/**
   374→ * Preview the audience for a broadcast WITHOUT mutating any state.
   375→ * Returns derived eligibility counts based on CURRENT contact consent state.
   376→ *
   377→ * DB-SIDE AGGREGATION: no N+1, no unbounded contact ID lists in memory.
   378→ * Uses a single GROUP BY query on Contact.marketingStatus, plus a single
   379→ * suppression count query that joins Contact ↔ SuppressionEntry.
   380→ *
   381→ * NOTE: preview eligibility may differ from send-time eligibility — consent
   382→ * can change between preview and actual send.
   383→ */
   384→export async function previewBroadcast(userId: number, broadcastId: string): Promise<PreviewResult | null> {
   385→  await requireBroadcastAccess(userId);
   386→  const broadcast = await db.broadcast.findFirst({
   387→    where: { broadcastId, userId },
   388→    select: { id: true, audienceType: true, targetGroupId: true },
   389→  });
   390→  if (!broadcast) return null;
   391→
   392→  // DB-side GROUP BY marketingStatus. For group audience, join ContactGroupMembership.
   393→  // We use $queryRawUnsafe with parameterized queries — never string interpolation.
   394→  const groupBySql = broadcast.audienceType === AUDIENCE_TYPES.GROUP
   395→    ? `SELECT c."marketingStatus" AS "marketingStatus", COUNT(*)::int AS cnt FROM "Contact" c
   396→        INNER JOIN "ContactGroupMembership" m ON m."contactId" = c."id" AND m."userId" = c."userId"
   397→        WHERE c."userId" = $1 AND m."groupId" = $2
   398→        GROUP BY c."marketingStatus"`
   399→    : `SELECT "marketingStatus", COUNT(*)::int AS cnt FROM "Contact"
   400→        WHERE "userId" = $1
   401→        GROUP BY "marketingStatus"`;
   402→
   403→  const statusRows: { marketingStatus: string; cnt: number }[] = broadcast.audienceType === AUDIENCE_TYPES.GROUP
   404→    ? await db.$queryRawUnsafe(groupBySql, userId, broadcast.targetGroupId)
   405→    : await db.$queryRawUnsafe(groupBySql, userId);
   406→
   407→  let total = 0;
   408→  let subscribedCount = 0;
   409→  let unsubscribed = 0;
   410→  let unknown = 0;       // unknown + any other
   411→  for (const row of statusRows) {
   412→    total += row.cnt;
   413→    if (row.marketingStatus === "subscribed") subscribedCount += row.cnt;
   414→    else if (row.marketingStatus === "unsubscribed") unsubscribed += row.cnt;
   415→    else unknown += row.cnt;
   416→  }
   417→
   418→  // DB-side count of active suppressions joined to the audience.
   419→  const suppressionSql = broadcast.audienceType === AUDIENCE_TYPES.GROUP
   420→    ? `SELECT COUNT(DISTINCT se."email")::int AS cnt FROM "SuppressionEntry" se
   421→        INNER JOIN "Contact" c ON c."email" = se."email" AND c."userId" = se."userId"
   422→        INNER JOIN "ContactGroupMembership" m ON m."contactId" = c."id" AND m."userId" = c."userId"
   423→        WHERE c."userId" = $1 AND m."groupId" = $2 AND se."active" = true AND c."marketingStatus" = 'subscribed' `
   424→    : `SELECT COUNT(DISTINCT se."email")::int AS cnt FROM "SuppressionEntry" se
   425→        INNER JOIN "Contact" c ON c."email" = se."email" AND c."userId" = se."userId"
   426→        WHERE c."userId" = $1 AND se."active" = true AND c."marketingStatus" = 'subscribed' `;
   427→
   428→  const supRows: { cnt: number }[] = broadcast.audienceType === AUDIENCE_TYPES.GROUP
   429→    ? await db.$queryRawUnsafe(suppressionSql, userId, broadcast.targetGroupId)
   430→    : await db.$queryRawUnsafe(suppressionSql, userId);
   431→  const suppressed = supRows[0]?.cnt ?? 0;
   432→
   433→  // MUTUAL EXCLUSION: suppressed contacts are subtracted from eligible.
   434→  // A subscribed + suppressed contact is counted as suppressed, NOT eligible.
   435→  // Categories are mutually exclusive: total = eligible + suppressed + unsubscribed + unknown.
   436→  const eligible = Math.max(0, subscribedCount - suppressed);
   437→
   438→  return { total, eligible, unknown, unsubscribed, suppressed };
   439→}

   441→// ---- Launch (DB-side INSERT...SELECT snapshot + freeze) -------------------
   442→
   443→/**
   444→ * Launch a broadcast: snapshot the audience (DB-side INSERT...SELECT), freeze
   445→ * content, check review threshold, and transition to queued (or review_pending
   446→ * if over threshold).
   447→ *
   448→ * Idempotent: if called again with the same broadcastId and the broadcast is
   449→ * already launched/snapshotted, returns the existing state without re-snapshotting.
   450→ *
   451→ * Idempotency-Key: if provided, durably dedupes across retries. Same key +
   452→ * different schedule → IdempotencyConflictError (409).
   453→ */
   454→export async function launchBroadcast(
   455→  userId: number,
   456→  broadcastId: string,
   457→  opts: { scheduledAt?: Date | null; idempotencyKey?: string } = {},
   458→): Promise<LaunchResult> {
   459→  await requireBroadcastAccess(userId);
   460→
   461→  const idempotencyKey = validateIdempotencyKey(opts.idempotencyKey);
   462→  const keyHash = idempotencyKey ? hashIdempotencyKey(userId, "launch", idempotencyKey) : null;
   463→  const fingerprint = idempotencyKey ? hashRequestFingerprint({ scheduledAt: opts.scheduledAt ?? null }) : null;
   464→
   465→  // Pre-check for replay (outside transaction — fast path).
   466→  if (keyHash) {
   467→    const existing = await db.broadcastMutationIdempotency.findUnique({
   468→      where: { userId_operation_idempotencyKeyHash: { userId, operation: "launch", idempotencyKeyHash: keyHash } },
   469→    });
   470→    if (existing) {
   471→      if (existing.targetBroadcastId !== broadcastId) {
   472→        throw new IdempotencyConflictError("Idempotency key reused for a different broadcast.");
   473→      }
   474→      if (existing.requestFingerprint && existing.requestFingerprint !== fingerprint) {
   475→        throw new IdempotencyConflictError("Idempotency key reused with conflicting schedule.");
   476→      }
   477→      // Replay the original outcome.
   478→      const data = existing.resultData as { status?: string; reviewStatus?: string; recipientCount?: number; requiresReview?: boolean; launched?: boolean } | null;
   479→      return {
   480→        launched: data?.launched ?? false,
   481→        broadcastId,
   482→        status: data?.status ?? BROADCAST_STATUSES.QUEUED,
   483→        reviewStatus: data?.reviewStatus ?? REVIEW_STATUSES.NOT_REQUIRED,
   484→        recipientCount: data?.recipientCount ?? 0,
   485→        requiresReview: data?.requiresReview ?? false,
   486→      };
   487→    }
   488→  }
   489→
   490→  const broadcast = await db.broadcast.findFirst({
   491→    where: { broadcastId, userId },
   492→    select: { id: true, status: true, audienceType: true, targetGroupId: true, subject: true, htmlContent: true, textContent: true },
   493→  });
   494→  if (!broadcast) {
   495→    throw new BroadcastValidationError("Broadcast not found.");
   496→  }
   497→
   498→  // Idempotent: already launched (no idempotency key or key not seen before).
   499→  if (broadcast.status !== BROADCAST_STATUSES.DRAFT) {
   500→    const current = await getBroadcast(userId, broadcastId);
   501→    return {
   502→      launched: false,
   503→      broadcastId,
   504→      status: current?.status ?? broadcast.status,
   505→      reviewStatus: current?.reviewStatus ?? REVIEW_STATUSES.NOT_REQUIRED,
   506→      recipientCount: current?.totalRecipients ?? 0,
   507→      requiresReview: current?.reviewStatus === REVIEW_STATUSES.PENDING,
   508→    };
   509→  }
   510→
   511→  // Validate content one more time before launch.
   512→  const contentValidation = validateBroadcastContent({
   513→    subject: broadcast.subject,
   514→    htmlContent: broadcast.htmlContent,
   515→    textContent: broadcast.textContent,
   516→  });
   517→  if (!contentValidation.valid) {
   518→    throw new BroadcastValidationError(contentValidation.error!);
   519→  }
   520→
   521→  // DB-side INSERT...SELECT snapshot inside the launch transaction.
   522→  // No full-audience ID array in Node memory. We then count the actual
   523→  // inserted rows for the review threshold check.
   524→  const reviewThreshold = BROADCAST_REVIEW_THRESHOLD;
   525→  const now = new Date();
   526→
   527→  try {
   528→    await db.$transaction(async (tx) => {
   529→      // Acquire advisory lock for concurrent idempotency serialization.
   530→      // This ensures two simultaneous calls with the SAME new key are serialized:
   531→      // the first wins the CAS + persists the idempotency record; the second
   532→      // sees the existing record inside the transaction and replays.
   533→      if (keyHash) {
   534→        // Derive a 64-bit lock key from (userId, "launch", keyHash).
   535→        // Use a hash to fit in a single bigint.
   536→        const lockKey = canonicalIdempotencyLockKey(userId, "launch", keyHash);
   537→        await tx.$executeRawUnsafe("SELECT pg_advisory_xact_lock($1::bigint)", lockKey.toString());
   538→
   539→        // Re-check idempotency INSIDE the transaction (after lock acquisition).
   540→        const existingInside = await tx.broadcastMutationIdempotency.findUnique({
   541→          where: { userId_operation_idempotencyKeyHash: { userId, operation: "launch", idempotencyKeyHash: keyHash } },
   542→        });
   543→        if (existingInside) {
   544→          if (existingInside.targetBroadcastId !== broadcastId) {
   545→            throw new IdempotencyConflictError("Idempotency key reused for a different broadcast.");
   546→          }
   547→          if (existingInside.requestFingerprint && existingInside.requestFingerprint !== fingerprint) {
   548→            throw new IdempotencyConflictError("Idempotency key reused with conflicting schedule.");
   549→          }
   550→          // Replay — the other transaction already launched.
   551→          const data = existingInside.resultData as { status?: string; reviewStatus?: string; recipientCount?: number; requiresReview?: boolean; launched?: boolean } | null;
   552→          return {
   553→            launched: data?.launched ?? false,
   554→            broadcastId,
   555→            status: data?.status ?? BROADCAST_STATUSES.QUEUED,
   556→            reviewStatus: data?.reviewStatus ?? REVIEW_STATUSES.NOT_REQUIRED,
   557→            recipientCount: data?.recipientCount ?? 0,
   558→            requiresReview: data?.requiresReview ?? false,
   559→          };
   560→        }
   561→      }
   562→
   563→      // CAS: only transition draft → queued/review_pending.
   564→      const claimed = await tx.broadcast.updateMany({
   565→        where: { id: broadcast.id, status: BROADCAST_STATUSES.DRAFT },
   566→        data: {
   567→          // Tentatively queued — we'll downgrade to review_pending if the count
   568→          // turns out to exceed the threshold.
   569→          status: BROADCAST_STATUSES.QUEUED,
   570→          reviewStatus: REVIEW_STATUSES.NOT_REQUIRED,
   571→          launchedAt: now,
   572→          scheduledAt: opts.scheduledAt ?? null,
   573→        },
   574→      });
   575→      if (claimed.count === 0) {
   576→        throw new BroadcastValidationError("Broadcast is no longer in draft status (concurrent launch).");
   577→      }
   578→
   579→      // DB-side audience snapshot via INSERT...SELECT...ON CONFLICT DO NOTHING.
   580→      if (broadcast.audienceType === AUDIENCE_TYPES.ALL_CONTACTS) {
   581→        await tx.$executeRawUnsafe(
   582→          `INSERT INTO "BroadcastRecipient" ("userId", "broadcastId", "contactOwnerUserId", "contactId", "status", "createdAt", "updatedAt")
   583→           SELECT $1, $2, "userId", "id", 'pending', NOW(), NOW() FROM "Contact"
   584→           WHERE "userId" = $1
   585→           ON CONFLICT DO NOTHING`,
   586→          userId,
   587→          broadcast.id,
   588→        );
   589→      } else if (broadcast.audienceType === AUDIENCE_TYPES.GROUP && broadcast.targetGroupId) {
   590→        await tx.$executeRawUnsafe(
   591→          `INSERT INTO "BroadcastRecipient" ("userId", "broadcastId", "contactOwnerUserId", "contactId", "status", "createdAt", "updatedAt")
   592→           SELECT $1, $2, m."userId", m."contactId", 'pending', NOW(), NOW()
   593→           FROM "ContactGroupMembership" m
   594→           INNER JOIN "Contact" c ON c."id" = m."contactId" AND c."userId" = m."userId"
   595→           WHERE m."userId" = $1 AND m."groupId" = $3
   596→           ON CONFLICT DO NOTHING`,
   597→          userId,
   598→          broadcast.id,
   599→          broadcast.targetGroupId,
   600→        );
   601→      }
   602→
   603→      // Count the actual inserted rows for the review threshold check.
   604→      const recipientCount = await tx.broadcastRecipient.count({ where: { broadcastId: broadcast.id } });
   605→      const requiresReview = recipientCount > reviewThreshold;
   606→
   607→      if (requiresReview) {
   608→        await tx.broadcast.update({
   609→          where: { id: broadcast.id },
   610→          data: {
   611→            status: BROADCAST_STATUSES.REVIEW_PENDING,
   612→            reviewStatus: REVIEW_STATUSES.PENDING,
   613→          },
   614→        });
   615→      }
   616→
   617→      // Persist the idempotency outcome (same transaction — atomic with state change).
   618→      if (idempotencyKey) {
   619→        const keyHash = hashIdempotencyKey(userId, "launch", idempotencyKey);
   620→        const fingerprint = hashRequestFingerprint({ scheduledAt: opts.scheduledAt ?? null });
   621→        try {
   622→          await tx.broadcastMutationIdempotency.create({
   623→            data: {
   624→              userId,
   625→              operation: "launch",
   626→              targetBroadcastId: broadcastId,
   627→              idempotencyKeyHash: keyHash,
   628→              requestFingerprint: fingerprint,
   629→              resultStatus: "applied",
   630→              resultData: {
   631→                launched: true,
   632→                status: requiresReview ? BROADCAST_STATUSES.REVIEW_PENDING : BROADCAST_STATUSES.QUEUED,
   633→                reviewStatus: requiresReview ? REVIEW_STATUSES.PENDING : REVIEW_STATUSES.NOT_REQUIRED,
   634→                recipientCount,
   635→                requiresReview,
   636→              },
   637→            },
   638→          });
   639→        } catch (err: any) {
   640→          // P2002 — concurrent same-key insert races us. The tx is now dead,
   641→          // so we let the error propagate OUT of $transaction; we'll handle
   642→          // it below by reading the existing record.
   643→          if (err?.code === "P2002") throw err;
   644→          throw err;
   645→        }
   646→      }
   647→    });
   648→  } catch (err: any) {
   649→    if (err?.code === "P2002" && idempotencyKey) {
   650→      // Concurrent idempotency insert race — read the existing record.
   651→      const keyHash = hashIdempotencyKey(userId, "launch", idempotencyKey);
   652→      const existing = await db.broadcastMutationIdempotency.findUnique({
   653→        where: { userId_operation_idempotencyKeyHash: { userId, operation: "launch", idempotencyKeyHash: keyHash } },
   654→      });
   655→      if (existing) {
   656→        if (existing.targetBroadcastId !== broadcastId) {
   657→          throw new IdempotencyConflictError("Idempotency key reused for a different broadcast.");
   658→        }
   659→        const data = existing.resultData as { status?: string; reviewStatus?: string; recipientCount?: number; requiresReview?: boolean; launched?: boolean } | null;
   660→        return {
   661→          launched: data?.launched ?? false,
   662→          broadcastId,
   663→          status: data?.status ?? BROADCAST_STATUSES.QUEUED,
   664→          reviewStatus: data?.reviewStatus ?? REVIEW_STATUSES.NOT_REQUIRED,
   665→          recipientCount: data?.recipientCount ?? 0,
   666→          requiresReview: data?.requiresReview ?? false,
   667→        };
   668→      }
   669→    }
   670→    throw err;
   671→  }
   672→
   673→  // Read back the final state to return accurate counts.
   674→  const finalBroadcast = await db.broadcast.findFirst({
   675→    where: { id: broadcast.id },
   676→    select: { status: true, reviewStatus: true },
   677→  });
   678→  const finalCount = await db.broadcastRecipient.count({ where: { broadcastId: broadcast.id } });
   679→  const requiresReview = finalCount > reviewThreshold;
   680→
   681→  return {
   682→    launched: true,
   683→    broadcastId,
   684→    status: finalBroadcast?.status ?? (requiresReview ? BROADCAST_STATUSES.REVIEW_PENDING : BROADCAST_STATUSES.QUEUED),
   685→    reviewStatus: finalBroadcast?.reviewStatus ?? (requiresReview ? REVIEW_STATUSES.PENDING : REVIEW_STATUSES.NOT_REQUIRED),
   686→    recipientCount: finalCount,
   687→    requiresReview,
   688→  };
   689→}

   691→// ---- Cancel (idempotent) --------------------------------------------------
   692→
   693→/**
   694→ * Cancel a broadcast. Idempotent — cancelling an already-cancelled broadcast
   695→ * is a no-op.
   696→ *
   697→ * Cancellable statuses: review_pending, queued, sending, paused_quota.
   698→ * Already-terminal statuses (completed, cancelled, rejected, failed) return
   699→ * the current state without error.
   700→ *
   701→ * Idempotency-Key: if provided, durably dedupes across retries.
   702→ */
   703→export async function cancelBroadcast(
   704→  userId: number,
   705→  broadcastId: string,
   706→  opts: { idempotencyKey?: string } = {},
   707→): Promise<{ cancelled: boolean; broadcastId: string; status: string }> {
   708→  await requireBroadcastAccess(userId);
   709→
   710→  const idempotencyKey = validateIdempotencyKey(opts.idempotencyKey);
   711→  const keyHash = idempotencyKey ? hashIdempotencyKey(userId, "cancel", idempotencyKey) : null;
   712→
   713→  if (idempotencyKey) {
   714→    const keyHash = hashIdempotencyKey(userId, "cancel", idempotencyKey);
   715→    const fingerprint = hashRequestFingerprint({});
   716→    const existing = await db.broadcastMutationIdempotency.findUnique({
   717→      where: { userId_operation_idempotencyKeyHash: { userId, operation: "cancel", idempotencyKeyHash: keyHash } },
   718→    });
   719→    if (existing) {
   720→      if (existing.targetBroadcastId !== broadcastId) {
   721→        throw new IdempotencyConflictError("Idempotency key reused for a different broadcast.");
   722→      }
   723→      // Ignore requestFingerprint for cancel — payload is empty.
   724→      const data = existing.resultData as { cancelled?: boolean; status?: string } | null;
   725→      return { cancelled: data?.cancelled ?? false, broadcastId, status: data?.status ?? BROADCAST_STATUSES.CANCELLED };
   726→    }
   727→  }
   728→
   729→  const broadcast = await db.broadcast.findFirst({
   730→    where: { broadcastId, userId },
   731→    select: { id: true, status: true },
   732→  });
   733→  if (!broadcast) {
   734→    throw new BroadcastValidationError("Broadcast not found.");
   735→  }
   736→
   737→  const now = new Date();
   738→  let cancelled = false;
   739→  let newStatus = broadcast.status;
   740→
   741→  // Everything happens inside ONE transaction with the advisory lock held.
   742→  // The idempotency outcome is persisted BEFORE commit, eliminating the
   743→  // post-lock race where Worker B could see no idempotency record while
   744→  // Worker A hasn't yet persisted it.
   745→  try {
   746→    await db.$transaction(async (tx) => {
   747→      if (keyHash) {
   748→        const lockKey = canonicalIdempotencyLockKey(userId, "cancel", keyHash);
   749→        await tx.$executeRawUnsafe("SELECT pg_advisory_xact_lock($1::bigint)", lockKey.toString());
   750→
   751→        // Re-check idempotency INSIDE the transaction (after lock acquisition).
   752→        const existingInside = await tx.broadcastMutationIdempotency.findUnique({
   753→          where: { userId_operation_idempotencyKeyHash: { userId, operation: "cancel", idempotencyKeyHash: keyHash } },
   754→        });
   755→        if (existingInside) {
   756→          if (existingInside.targetBroadcastId !== broadcastId) {
   757→            throw new IdempotencyConflictError("Idempotency key reused for a different broadcast.");
   758→          }
   759→          const data = existingInside.resultData as { cancelled?: boolean; status?: string } | null;
   760→          cancelled = data?.cancelled ?? false;
   761→          newStatus = data?.status ?? BROADCAST_STATUSES.CANCELLED;
   762→          return;
   763→        }
   764→      }
   765→
   766→      // Evaluate current Broadcast state inside the locked transaction.
   767→      const currentBroadcast = await tx.broadcast.findFirst({
   768→        where: { id: broadcast.id },
   769→        select: { status: true },
   770→      });
   771→      if (!currentBroadcast || !CANCELLABLE_STATUSES.has(currentBroadcast.status)) {
   772→        // Already terminal — no-op. Persist the no-op outcome under the lock.
   773→        cancelled = false;
   774→        newStatus = currentBroadcast?.status ?? broadcast.status;
   775→      } else {
   776→        const result = await tx.broadcast.updateMany({
   777→          where: { id: broadcast.id, status: { in: [...CANCELLABLE_STATUSES] } },
   778→          data: {
   779→            status: BROADCAST_STATUSES.CANCELLED,
   780→            cancelledAt: now,
   781→          },
   782→        });
   783→        cancelled = result.count === 1;
   784→        newStatus = cancelled ? BROADCAST_STATUSES.CANCELLED : currentBroadcast.status;
   785→      }
   786→
   787→      // Persist idempotency outcome INSIDE the same transaction, while the lock is held.
   788→      if (keyHash) {
   789→        await tx.broadcastMutationIdempotency.create({
   790→          data: {
   791→            userId,
   792→            operation: "cancel",
   793→            targetBroadcastId: broadcastId,
   794→            idempotencyKeyHash: keyHash,
   795→            requestFingerprint: hashRequestFingerprint({}),
   796→            resultStatus: "applied",
   797→            resultData: { cancelled, status: newStatus },
   798→          },
   799→        });
   800→      }
   801→    });
   802→  } catch (err: any) {
   803→    if (err instanceof IdempotencyConflictError) throw err;
   804→    // P2002 means a concurrent cancel with the same key already committed
   805→    // — re-read and replay.
   806→    if (err?.code === "P2002" && keyHash) {
   807→      const existing = await db.broadcastMutationIdempotency.findUnique({
   808→        where: { userId_operation_idempotencyKeyHash: { userId, operation: "cancel", idempotencyKeyHash: keyHash } },
   809→      });
   810→      if (existing) {
   811→        const data = existing.resultData as { cancelled?: boolean; status?: string } | null;
   812→        return { cancelled: data?.cancelled ?? false, broadcastId, status: data?.status ?? BROADCAST_STATUSES.CANCELLED };
   813→      }
   814→    }
   815→    throw err;
   816→  }
   817→
   818→  return { cancelled, broadcastId, status: newStatus };
   819→}

   822→// ---- Recipient claiming (atomic CAS) --------------------------------------
   823→
   824→/**
   825→ * Atomically claim a batch of pending recipients for processing.
   826→ * Uses CAS (updateMany WHERE status=pending) + row-level locking.
   827→ *
   828→ * Only successfully claimed rows may be processed. Stale locks are recovered
   829→ * by `recoverStaleRecipients()` (processing only — NOT dispatching).
   830→ */
   831→export async function claimRecipientBatch(broadcastId: number, workerId: string, batchSize: number = BROADCAST_BATCH_SIZE): Promise<number[]> {
   832→  const candidates = await db.broadcastRecipient.findMany({
   833→    where: { broadcastId, status: RECIPIENT_STATUSES.PENDING },
   834→    orderBy: { id: "asc" },
   835→    take: batchSize,
   836→    select: { id: true },
   837→  });
   838→
   839→  const claimedIds: number[] = [];
   840→  const now = new Date();
   841→  for (const candidate of candidates) {
   842→    const result = await db.broadcastRecipient.updateMany({
   843→      where: { id: candidate.id, status: RECIPIENT_STATUSES.PENDING },
   844→      data: {
   845→        status: RECIPIENT_STATUSES.PROCESSING,
   846→        lockedAt: now,
   847→        lockedBy: workerId,
   848→      },
   849→    });
   850→    if (result.count === 1) {
   851→      claimedIds.push(candidate.id);
   852→    }
   853→  }
   854→  return claimedIds;
   855→}
   856→
   857→/**
   858→ * Recover stale PROCESSING claims: processing → pending WHERE lockedAt < cutoff.
   859→ * A stale worker must NEVER overwrite a newer worker's terminal result.
   860→ *
   861→ * This function ONLY recovers `processing` rows — it MUST NOT recover
   862→ * `dispatching` rows. Dispatching rows have entered the durable exclusive
   863→ * dispatch state and may have called the provider — auto-requeue could
   864→ * double-send. Stale dispatching rows are handled by recoverAbandonedDispatches().
   865→ */
   866→export async function recoverStaleRecipients(): Promise<number> {
   867→  const cutoff = new Date(Date.now() - BROADCAST_STALE_LOCK_TIMEOUT_MS);
   868→  const stale = await db.broadcastRecipient.findMany({
   869→    where: { status: RECIPIENT_STATUSES.PROCESSING, lockedAt: { lt: cutoff } },
   870→    select: { id: true },
   871→  });
   872→  let recovered = 0;
   873→  for (const row of stale) {
   874→    const result = await db.broadcastRecipient.updateMany({
   875→      where: { id: row.id, status: RECIPIENT_STATUSES.PROCESSING, lockedAt: { lt: cutoff } },
   876→      data: {
   877→        status: RECIPIENT_STATUSES.PENDING,
   878→        lockedAt: null,
   879→        lockedBy: null,
   880→      },
   881→    });
   882→    if (result.count === 1) recovered++;
   883→  }
   884→  return recovered;
   885→}
   886→
   887→/**
   888→ * Recover abandoned DISPATCHING rows: dispatching → failed WHERE lockedAt <
   889→ * dispatchTimeout. NEVER auto-requeue — the outcome is ambiguous (provider
   890→ * may have been called). Transition to terminal `failed` with
   891→ * errorCode=provider_outcome_unknown.
   892→ *
   893→ * This function is SEPARATE from recoverStaleRecipients() and uses a longer
   894→ * timeout (30 min) to give even slow providers time to complete.
   895→ */
   896→export async function recoverAbandonedDispatches(): Promise<number> {
   897→  const cutoff = new Date(Date.now() - BROADCAST_DISPATCH_TIMEOUT_MS);
   898→  const stale = await db.broadcastRecipient.findMany({
   899→    where: { status: RECIPIENT_STATUSES.DISPATCHING, lockedAt: { lt: cutoff } },
   900→    select: { id: true },
   901→  });
   902→  let recovered = 0;
   903→  const now = new Date();
   904→  for (const row of stale) {
   905→    const result = await db.broadcastRecipient.updateMany({
   906→      where: { id: row.id, status: RECIPIENT_STATUSES.DISPATCHING, lockedAt: { lt: cutoff } },
   907→      data: {
   908→        status: RECIPIENT_STATUSES.FAILED,
   909→        errorCode: SEND_ERROR_CODES.PROVIDER_OUTCOME_UNKNOWN,
   910→        attemptedAt: now,
   911→        failedAt: now,
   912→        // Leave lockedBy/lockedAt intact for audit — the row is now terminal.
   913→      },
   914→    });
   915→    if (result.count === 1) recovered++;
   916→  }
   917→  return recovered;
   918→}

   920→// ---- Send processing ------------------------------------------------------
   921→
   922→export interface ProcessResult {
   923→  processed: number;
   924→  sent: number;
   925→  skipped: number;
   926→  failed: number;
   927→  quotaPaused: boolean;
   928→  /**
   929→   * True iff the broadcast transitioned to `completed` during this invocation.
   930→   * Cron accounting counts this flag, NOT `processed === 0`.
   931→   */
   932→  finalized: boolean;
   933→}
   934→
   935→/**
   936→ * Process a single broadcast: claim a batch, send each recipient, finalize.
   937→ *
   938→ * Per-recipient flow:
   939→ *   1. Verify parent broadcast not cancelled.
   940→ *   2. Re-read Contact (may have been deleted → contactId null).
   941→ *   3. Check Phase 9 marketing eligibility (subscribed AND not suppressed).
   942→ *   4. Generate per-recipient unsubscribe token (JWE).
   943→ *   5. Render content with per-recipient variables.
   944→ *   6. CAS processing → dispatching WHERE lockedBy=workerId (durable exclusive dispatch).
   945→ *   7. Reserve BROADCAST_EMAILS quota (atomic CAS) — AFTER dispatching CAS but
   946→ *      BEFORE provider call.
   947→ *      - If quota denied with reason=quota_exhausted → pause broadcast +
   948→ *        transition dispatching → pending (no external attempt occurred).
   949→ *      - If quota denied with reason=rate_limited → recipient stays pending,
   950→ *        broadcast stays in `sending` state.
   951→ *      - DB/entitlement errors → terminal `failed` with safe error code.
   952→ *   8. Dispatch via provider abstraction.
   953→ *   9. Terminal CAS: dispatching → sent|skipped|failed WHERE lockedBy=workerId.
   954→ *
   955→ * Quota: skipped recipients consume NO quota. A real provider attempt
   956→ * consumes exactly 1. Idempotent replays consume no second quota.
   957→ */
   958→export async function processBroadcast(broadcastId: number, provider?: EmailProvider): Promise<ProcessResult> {
   959→  const result: ProcessResult = { processed: 0, sent: 0, skipped: 0, failed: 0, quotaPaused: false, finalized: false };
   960→
   961→  // Recover stale processing rows (NOT dispatching — those are ambiguous).
   962→  await recoverStaleRecipients();
   963→  // Recover abandoned dispatching rows (terminal failed, no auto-requeue).
   964→  await recoverAbandonedDispatches();
   965→
   966→  const workerId = randomUUID();
   967→  const broadcast = await db.broadcast.findUnique({
   968→    where: { id: broadcastId },
   969→    select: { id: true, userId: true, status: true, scheduledAt: true, subject: true, htmlContent: true, textContent: true },
   970→  });
   971→  if (!broadcast) return result;
   972→
   973→  // Don't process cancelled broadcasts.
   974→  if (broadcast.status === BROADCAST_STATUSES.CANCELLED) return result;
   975→
   976→  // Don't process review_pending or rejected broadcasts.
   977→  if (broadcast.status === BROADCAST_STATUSES.REVIEW_PENDING || broadcast.status === BROADCAST_STATUSES.REJECTED) return result;
   978→
   979→  // Don't process future-scheduled broadcasts.
   980→  if (broadcast.scheduledAt && broadcast.scheduledAt.getTime() > Date.now()) return result;
   981→
   982→  // Transition queued → sending (CAS).
   983→  if (broadcast.status === BROADCAST_STATUSES.QUEUED || broadcast.status === BROADCAST_STATUSES.PAUSED_QUOTA) {
   984→    await db.broadcast.updateMany({
   985→      where: { id: broadcast.id, status: { in: [BROADCAST_STATUSES.QUEUED, BROADCAST_STATUSES.PAUSED_QUOTA] } },
   986→      data: { status: BROADCAST_STATUSES.SENDING, startedAt: new Date() },
   987→    });
   988→  }
   989→
   990→  // Claim a batch.
   991→  const recipientIds = await claimRecipientBatch(broadcast.id, workerId, BROADCAST_BATCH_SIZE);
   992→  if (recipientIds.length === 0) {
   993→    // Maybe finalize.
   994→    result.finalized = await tryFinalizeBroadcast(broadcast.id, workerId);
   995→    return result;
   996→  }
   997→
   998→  const emailProvider = provider ?? new SmtpEmailProvider();
   999→
  1000→  for (const recipientId of recipientIds) {
  1001→    result.processed++;
  1002→    await processRecipient(broadcast, recipientId, workerId, emailProvider, result);
  1003→  }
  1004→
  1005→  // Try to finalize after the batch.
  1006→  result.finalized = await tryFinalizeBroadcast(broadcast.id, workerId);
  1007→
  1008→  return result;
  1009→}
  1010→
  1011→async function processRecipient(
  1012→  broadcast: { id: number; userId: number; subject: string; htmlContent: string; textContent: string | null },
  1013→  recipientId: number,
  1014→  workerId: string,
  1015→  provider: EmailProvider,
  1016→  result: ProcessResult,
  1017→): Promise<void> {
  1018→  const recipient = await db.broadcastRecipient.findUnique({
  1019→    where: { id: recipientId },
  1020→    select: { id: true, userId: true, broadcastId: true, contactId: true, contactOwnerUserId: true, status: true, lockedBy: true },
  1021→  });
  1022→  if (!recipient || recipient.status !== RECIPIENT_STATUSES.PROCESSING || recipient.lockedBy !== workerId) {
  1023→    return; // Lost ownership — stale worker.
  1024→  }
  1025→
  1026→  // Re-check parent broadcast state.
  1027→  const parent = await db.broadcast.findUnique({
  1028→    where: { id: broadcast.id },
  1029→    select: { status: true },
  1030→  });
  1031→  if (parent?.status === BROADCAST_STATUSES.CANCELLED) {
  1032→    const won = await markRecipientSkipped(recipientId, workerId, SKIP_REASONS.BROADCAST_CANCELLED);
  1033→    if (won) result.skipped++;
  1034→    return;
  1035→  }
  1036→
  1037→  // Handle null Contact reference (Contact was deleted after snapshot).
  1038→  // The composite FK nullified (contactOwnerUserId, contactId) on Contact deletion.
  1039→  // CHECK constraint ensures both are null or both are present — one check suffices.
  1040→  if (recipient.contactId === null || recipient.contactOwnerUserId === null) {
  1041→    const won2 = await markRecipientSkipped(recipientId, workerId, SKIP_REASONS.CONTACT_NOT_FOUND);
  1042→    if (won2) result.skipped++;
  1043→    return;
  1044→  }
  1045→
  1046→  const contact = await db.contact.findFirst({
  1047→    where: { id: recipient.contactId, userId: recipient.contactOwnerUserId },
  1048→    select: { id: true, email: true, name: true },
  1049→  });
  1050→  if (!contact) {
  1051→    const won = await markRecipientSkipped(recipientId, workerId, SKIP_REASONS.CONTACT_NOT_FOUND);
  1052→    if (won) result.skipped++;
  1053→    return;
  1054→  }
  1055→
  1056→  // Check Phase 9 marketing eligibility immediately before dispatch.
  1057→  const eligibility = await getMarketingEligibility(recipient.userId, recipient.contactId);
  1058→  if (!eligibility.eligible) {
  1059→    const reason = eligibility.reason === "suppressed" ? SKIP_REASONS.SUPPRESSED : SKIP_REASONS.NOT_SUBSCRIBED;
  1060→    const won3 = await markRecipientSkipped(recipientId, workerId, reason);
  1061→    if (won3) result.skipped++;
  1062→    return;
  1063→  }
  1064→
  1065→  // Generate per-recipient unsubscribe token (JWE).
  1066→  const token = await mintUnsubscribeToken({
  1067→    userId: recipient.userId,
  1068→    contactId: contact.id,
  1069→    email: contact.email,
  1070→  });
  1071→  const unsubscribeUrl = buildUnsubscribeUrl(token);
  1072→  const listUnsubscribeHeader = buildListUnsubscribeHeader(token);
  1073→
  1074→  // Render content with per-recipient variables + final sanitization.
  1075→  const rendered = renderBroadcastContent({
  1076→    subject: broadcast.subject,
  1077→    htmlContent: broadcast.htmlContent,
  1078→    textContent: broadcast.textContent,
  1079→    contactId: contact.id,
  1080→    contactEmail: contact.email,
  1081→    contactName: contact.name,
  1082→    unsubscribeUrl,
  1083→  });
  1084→  if (!rendered.ok) {
  1085→    const won4 = await markRecipientFailed(recipientId, workerId, SEND_ERROR_CODES.CONTENT_RENDER_ERROR);
  1086→    if (won4) result.failed++;
  1087→    return;
  1088→  }
  1089→
  1090→  // Reserve BROADCAST_EMAILS quota (atomic CAS) AFTER the dispatching CAS but
  1091→  // BEFORE the provider call.
  1092→  //
  1093→  // Order rationale:
  1094→  //   1. CAS processing → dispatching WHERE lockedBy=workerId (durable exclusive
  1095→  //      dispatch state). ONLY the winner of this CAS may proceed.
  1096→  //   2. checkUsage() (atomic CAS — consumes quota if allowed). If the winner
  1097→  //      is denied quota, it transitions dispatching → pending safely (no
  1098→  //      external attempt occurred).
  1099→  //   3. provider.send() — only if both 1 and 2 succeed.
  1100→  //
  1101→  // Stale-worker safety:
  1102→  //   - The dispatching CAS uniquely assigns the recipient to one worker. A
  1103→  //     stale worker CANNOT re-enter dispatching (the row is already there).
  1104→  //   - recoverStaleRecipients() does NOT touch dispatching rows.
  1105→  //   - recoverAbandonedDispatches() transitions stale dispatching → terminal
  1106→  //     `failed` with errorCode=provider_outcome_unknown — never to pending.
  1107→  //
  1108→  // Quota correctness:
  1109→  //   - Only the dispatching winner consumes quota (the loser doesn't even
  1110→  //     reach the checkUsage call).
  1111→  //   - If quota is denied AFTER the CAS, the winner transitions dispatching →
  1112→  //     pending safely (no provider call) — the recipient is re-claimable
  1113→  //     when quota resets.
  1114→
  1115→  // CAS processing → dispatching WHERE lockedBy=workerId (durable exclusive dispatch).
  1116→  // ONLY the worker that wins this CAS may call provider.send().
  1117→  // Refresh lockedAt so the dispatch timeout measures from dispatch start,
  1118→  // not from the earlier processing claim.
  1119→  const dispatchingCas = await db.broadcastRecipient.updateMany({
  1120→    where: { id: recipientId, status: RECIPIENT_STATUSES.PROCESSING, lockedBy: workerId },
  1121→    data: {
  1122→      status: RECIPIENT_STATUSES.DISPATCHING,
  1123→      lockedAt: new Date(),
  1124→    },
  1125→  });
  1126→  if (dispatchingCas.count !== 1) {
  1127→    // Lost the CAS — another worker already took over, or row is no longer in
  1128→    // processing state. This worker must NOT consume quota or call provider.send().
  1129→    return;
  1130→  }
  1131→
  1132→  // Re-check parent cancellation AFTER winning dispatching CAS, BEFORE quota/provider.
  1133→  // This minimizes the cancellation race window.
  1134→  const parentAfterDispatch = await db.broadcast.findUnique({
  1135→    where: { id: broadcast.id },
  1136→    select: { status: true },
  1137→  });
  1138→  if (parentAfterDispatch?.status === BROADCAST_STATUSES.CANCELLED) {
  1139→    const won = await markRecipientSkippedFromDispatching(recipientId, workerId, SKIP_REASONS.BROADCAST_CANCELLED);
  1140→    if (won) result.skipped++;
  1141→    return;
  1142→  }
  1143→
  1144→  // We are now the exclusive dispatcher. Consume quota.
  1145→  const usage = await checkUsage(recipient.userId, FEATURE_KEYS.BROADCAST_EMAILS);
  1146→  if (!usage.allowed) {
  1147→    if (usage.reason === "quota_exhausted") {
  1148→      // Quota exhausted — pause the broadcast, transition dispatching → pending
  1149→      // (no provider call has happened, so safe to re-queue).
  1150→      // Only set quotaPaused if the Broadcast actually transitions sending → paused_quota.
  1151→      const pauseResult = await db.broadcast.updateMany({
  1152→        where: { id: broadcast.id, status: BROADCAST_STATUSES.SENDING },
  1153→        data: { status: BROADCAST_STATUSES.PAUSED_QUOTA },
  1154→      });
  1155→      await db.broadcastRecipient.updateMany({
  1156→        where: { id: recipientId, status: RECIPIENT_STATUSES.DISPATCHING, lockedBy: workerId },
  1157→        data: {
  1158→          status: RECIPIENT_STATUSES.PENDING,
  1159→          lockedAt: null,
  1160→          lockedBy: null,
  1161→        },
  1162→      });
  1163→      // Set quotaPaused only if we actually won the sending → paused_quota CAS,
  1164→      // OR if a re-read confirms the Broadcast is already paused for quota.
  1165→      if (pauseResult.count === 1) {
  1166→        result.quotaPaused = true;
  1167→      } else {
  1168→        const recheck = await db.broadcast.findUnique({ where: { id: broadcast.id }, select: { status: true } });
  1169→        if (recheck?.status === BROADCAST_STATUSES.PAUSED_QUOTA) {
  1170→          result.quotaPaused = true;
  1171→        }
  1172→      }
  1173→      return;
  1174→    }
  1175→    if (usage.reason === "rate_limited") {
  1176→      // Rate limited — recipient stays pending, broadcast stays in `sending` state.
  1177→      // No provider call, no quota consumed (rate limit is checked before quota
  1178→      // consume in checkUsage()).
  1179→      await db.broadcastRecipient.updateMany({
  1180→        where: { id: recipientId, status: RECIPIENT_STATUSES.DISPATCHING, lockedBy: workerId },
  1181→        data: {
  1182→          status: RECIPIENT_STATUSES.PENDING,
  1183→          lockedAt: null,
  1184→          lockedBy: null,
  1185→        },
  1186→      });
  1187→      return;
  1188→    }
  1189→    // DB/entitlement error — terminal `failed` with safe error code.
  1190→    const quotaFailWon = await markRecipientFailedFromDispatching(recipientId, workerId, SEND_ERROR_CODES.QUOTA_ERROR);
  1191→    if (quotaFailWon) result.failed++;
  1192→    return;
  1193→  }
  1194→
  1195→  // Dispatch via provider. Stale from here → recoverAbandonedDispatches.
  1196→  const now = new Date();
  1197→  try {
  1198→    const sendResult = await provider.send({
  1199→      to: contact.email,
  1200→      subject: rendered.subject,
  1201→      html: rendered.html,
  1202→      text: rendered.text,
  1203→      headers: {
  1204→        "List-Unsubscribe": listUnsubscribeHeader,
  1205→        "List-Unsubscribe-Post": "List-Unsubscribe=One-Click",
  1206→      },
  1207→    });
  1208→
  1209→    // Terminal CAS: dispatching → sent WHERE lockedBy=workerId.
  1210→    // Only increment counter if the CAS actually succeeds (count === 1).
  1211→    // If the CAS fails, the external email may already have been delivered,
  1212→    // but this worker did NOT record the terminal state — do NOT increment.
  1213→    const sentResult = await db.broadcastRecipient.updateMany({
  1214→      where: { id: recipientId, status: RECIPIENT_STATUSES.DISPATCHING, lockedBy: workerId },
  1215→      data: {
  1216→        status: RECIPIENT_STATUSES.SENT,
  1217→        providerMessageId: sendResult.messageId ?? null,
  1218→        attemptedAt: now,
  1219→        sentAt: now,
  1220→        lockedAt: null,
  1221→        lockedBy: null,
  1222→      },
  1223→    });
  1224→    if (sentResult.count === 1) result.sent++;
  1225→  } catch (err: any) {
  1226→    const errorCode = classifySendError(err);
  1227→    const failWon = await markRecipientFailedFromDispatching(recipientId, workerId, errorCode);
  1228→    if (failWon) result.failed++;
  1229→  }
  1230→}
  1231→
  1232→function classifySendError(err: any): string {
  1233→  const msg = err?.message ?? "";
  1234→  if (msg.includes("configuration") || msg.includes("EAUTH") || msg.includes("EENVELOPE")) {
  1235→    return SEND_ERROR_CODES.CONFIGURATION_ERROR;
  1236→  }
  1237→  if (msg.includes("timeout") || msg.includes("ECONNECTION")) {
  1238→    return SEND_ERROR_CODES.PROVIDER_ERROR;
  1239→  }
  1240→  return SEND_ERROR_CODES.PROVIDER_ERROR;
  1241→}
