     1→/**
     2→ * Consent & Suppression — central service layer (Phase 9, audit revision).
     3→ *
     4→ * DESIGN INVARIANTS (per Phase 9 audit):
     5→ *
     6→ * 1. ONE central service for ALL consent/suppression mutations. Routes must
     7→ *    NOT independently mutate Contact.marketing* fields or SuppressionEntry.
     8→ *
     9→ * 2. CANONICAL MUTATION LOCK: every consent/suppression mutation for a given
    10→ *    (userId, normalized email) acquires a transaction-scoped PostgreSQL
    11→ *    advisory lock via `pg_advisory_xact_lock(canonical_key)`. This
    12→ *    serializes concurrent operations on the same target. Different tenants
    13→ *    have independent lock domains. The lock is held for the duration of the
    14→ *    `db.$transaction()` call.
    15→ *
    16→ * 3. NO in-transaction catch-and-continue: a P2002 or any other constraint
    17→ *    error inside `db.$transaction()` invalidates the PostgreSQL transaction.
    18→ *    We NEVER `try/catch` constraint errors inside the tx and continue — we
    19→ *    let them propagate. P2002 from concurrent idempotent inserts is caught
    20→ *    OUTSIDE the tx and resolved by reading the existing event row.
    21→ *
    22→ * 4. IDEMPOTENCY NAMESPACE: idempotency is bound to
    23→ *    `(userId, operation, target, idempotencyKeyHash)`. The same caller key
    24→ *    reused for a DIFFERENT operation or contact does NOT replay an unrelated
    25→ *    prior result. A separate `requestFingerprint` (SHA-256 of the mutable
    26→ *    request payload) detects same-key-different-body conflicts → 409
    27→ *    `idempotency_conflict` at the service boundary (thrown as
    28→ *    `IdempotencyConflictError`).
    29→ *
    30→ * 5. NO-OP SEMANTICS: a same-state repeated request (e.g. subscribe when
    31→ *    already subscribed, no idempotency key) returns `status: "no_op"` and
    32→ *    does NOT create a fake transition event. Only ACTUAL state transitions
    33→ *    append to the immutable audit history. Retries are handled by the
    34→ *    idempotency key — they return `idempotent_replay` and dedupe.
    35→ *
    36→ * 6. STRUCTURAL TENANT ISOLATION: composite FKs
    37→ *    (ContactConsentEvent.userId, contactId) → Contact(userId, id) and
    38→ *    (SuppressionEvent.userId, suppressionId) → SuppressionEntry(userId, id)
    39→ *    enforce tenant agreement at the DB level. Application-level filters are
    40→ *    defense-in-depth, not the primary enforcement.
    41→ *
    42→ * 7. Contact PATCH and other unrelated mutations MUST NOT touch marketing
    43→ *    fields. Only this service's subscribe/unsubscribe operations change
    44→ *    marketingStatus.
    45→ *
    46→ * 8. Transactional Send is NOT gated by marketing eligibility — only future
    47→ *    Phase 10 Broadcast sending will consult getMarketingEligibility.
    48→ */

   111→// ---- Idempotency hashing ---------------------------------------------------
   112→
   113→const IDEMPOTENCY_DOMAIN = "nixify:phase9:consent:v1";
   114→
   115→/**
   116→ * Hash an idempotency key with a tenant-bound domain separator.
   117→ *
   118→ * The hash is what we store — never the raw key. This prevents a leaked DB
   119→ * row from being usable to replay requests, and ensures cross-tenant
   120→ * collisions are impossible even if two tenants happen to use the same key.
   121→ *
   122→ * NOTE: the (userId, operation, idempotencyKeyHash) tuple is the canonical
   123→ * idempotency namespace. The operation field on the audit row ensures the
   124→ * same caller key cannot replay an unrelated prior operation. The target
   125→ * (contactId/email) is verified to match the existing event before replay.
   126→ */
   127→export function hashIdempotencyKey(userId: number, operation: string, key: string): string {
   128→  return createHash("sha256")
   129→    .update(`${IDEMPOTENCY_DOMAIN}:uid=${userId}:op=${operation}:key=${key}`)
   130→    .digest("hex");
   131→}
   132→
   133→/**
   134→ * Hash the mutable request payload to detect same-key-different-body conflicts.
   135→ * The hash is stored as `requestFingerprint`. If a retried request has the
   136→ * same idempotency key but a different fingerprint, we throw
   137→ * IdempotencyConflictError → 409 at the route boundary.
   138→ */
   139→export function hashRequestFingerprint(payload: unknown): string {
   140→  // Stable canonicalization: sort object keys, then JSON-stringify.
   141→  const stable = JSON.stringify(payload, (_k, v) =>
   142→    v !== null && typeof v === "object" && !Array.isArray(v)
   143→      ? Object.keys(v).sort().reduce<Record<string, unknown>>((acc, k) => { acc[k] = (v as Record<string, unknown>)[k]; return acc; }, {})
   144→      : v
   145→  );
   146→  return createHash("sha256").update(stable).digest("hex");
   147→}

   149→// ---- Canonical mutation lock ----------------------------------------------
   150→
   151→/**
   152→ * Stable 64-bit canonical mutation key derived from (userId, normalized email).
   153→ *
   154→ * PostgreSQL has TWO `pg_advisory_xact_lock` overloads:
   155→ *   - pg_advisory_xact_lock(key bigint) — single 64-bit key
   156→ *   - pg_advisory_xact_lock(key1 integer, key2 integer) — two 32-bit keys
   157→ * Passing two bigints matches NEITHER overload (PostgreSQL does not
   158→ * auto-cast bigint → integer for function resolution). We use the single-
   159→ * bigint overload and pack (tenant, email) into one 64-bit key:
   160→ *   high 32 bits = userId (tenant isolation — different tenants → different
   161→ *                  high halves → independent lock domains)
   162→ *   low 32 bits  = first 8 hex chars of SHA-256(normalizedEmail) as uint32
   163→ *                  (same email → same low half regardless of caller)
   164→ */
   165→function canonicalLockKey(userId: number, normalizedEmail: string): bigint {
   166→  const tenant = BigInt(userId) & BigInt("0xffffffff"); // low 32 bits of userId
   167→  const emailHex = createHash("sha256").update(normalizedEmail).digest("hex").slice(0, 8);
   168→  const email = BigInt(parseInt(emailHex, 16)) & BigInt("0xffffffff");
   169→  // Pack: high 32 bits = tenant, low 32 bits = email hash.
   170→  return (tenant << BigInt(32)) | email;
   171→}
   172→
   173→/**
   174→ * Acquire a transaction-scoped advisory lock for the canonical (userId, email)
   175→ * mutation target. The lock is automatically released when the transaction
   176→ * commits or rolls back.
   177→ *
   178→ * This serializes concurrent consent/suppression operations on the same
   179→ * (tenant, email) pair, ensuring the post-lock state read is fresh and the
   180→ * history transition chain is coherent.
   181→ *
   182→ * Uses $executeRawUnsafe with a parameterized query — passing the bigint as
   183→ * a string avoids Prisma's BigInt serialization quirks. The cast to ::bigint
   184→ * is explicit so PostgreSQL resolves the function overload unambiguously.
   185→ */
   186→async function acquireCanonicalLock(
   187→  tx: Parameters<Parameters<typeof db.$transaction>[0]>[0],
   188→  userId: number,
   189→  normalizedEmail: string,
   190→): Promise<void> {
   191→  const key = canonicalLockKey(userId, normalizedEmail);
   192→  await tx.$executeRawUnsafe("SELECT pg_advisory_xact_lock($1::bigint)", key.toString());
   193→}

   293→// ---- Internal: idempotency record helpers --------------------------------
   294→
   295→/**
   296→ * Durable idempotency outcome record lookup.
   297→ *
   298→ * Decoupled from transition-event tables so that a NO-OP request can be
   299→ * durably replayed. A retry with the same key (even after the contact's
   300→ * state has changed) replays the ORIGINAL outcome — it does NOT re-evaluate
   301→ * and potentially apply a transition that the original request did not apply.
   302→ *
   303→ * The unique namespace is (userId, operation, idempotencyKeyHash). The
   304→ * target (targetType + targetKey) is verified before replay to prevent
   305→ * cross-target key reuse. The requestFingerprint is verified to detect
   306→ * same-key-different-payload conflicts.
   307→ *
   308→ * Returns null when:
   309→ *   - no idempotencyKeyHash was supplied (no dedup requested), OR
   310→ *   - no prior record exists for this (userId, operation, idempotencyKeyHash).
   311→ */
   312→async function findExistingIdempotency(
   313→  tx: Parameters<Parameters<typeof db.$transaction>[0]>[0],
   314→  userId: number,
   315→  operation: string,
   316→  idempotencyKeyHash: string | null,
   317→  expectedTargetType: string,
   318→  expectedTargetKey: string,
   319→): Promise<{
   320→  requestFingerprint: string | null;
   321→  resultStatus: string;
   322→  resultEventId: string | null;
   323→  resultSuppressionId: string | null;
   324→} | null> {
   325→  if (!idempotencyKeyHash) return null;
   326→  const existing = await tx.consentMutationIdempotency.findUnique({
   327→    where: { userId_operation_idempotencyKeyHash: { userId, operation, idempotencyKeyHash } },
   328→    select: {
   329→      requestFingerprint: true,
   330→      resultStatus: true,
   331→      resultEventId: true,
   332→      resultSuppressionId: true,
   333→      targetType: true,
   334→      targetKey: true,
   335→    },
   336→  });
   337→  if (!existing) return null;
   338→  // Verify the target matches — same key reused for a different target is a conflict.
   339→  if (existing.targetType !== expectedTargetType || existing.targetKey !== expectedTargetKey) {
   340→    throw new IdempotencyConflictError(
   341→      `Idempotency key already used for a different ${expectedTargetType} target.`,
   342→    );
   343→  }
   344→  return {
   345→    requestFingerprint: existing.requestFingerprint,
   346→    resultStatus: existing.resultStatus,
   347→    resultEventId: existing.resultEventId,
   348→    resultSuppressionId: existing.resultSuppressionId,
   349→  };
   350→}
   351→
   352→/**
   353→ * Persist a durable idempotency outcome record. Called after the mutation
   354→ * (or no-op) has been decided but BEFORE the transaction commits, so the
   355→ * outcome record and any state mutation commit atomically.
   356→ *
   357→ * If the (userId, operation, idempotencyKeyHash) tuple already exists (race
   358→ * with a concurrent insert), P2002 propagates out of the transaction and is
   359→ * caught by the outer try/catch — which then re-reads and returns the
   360→ * existing outcome.
   361→ */
   362→async function persistIdempotencyOutcome(
   363→  tx: Parameters<Parameters<typeof db.$transaction>[0]>[0],
   364→  opts: {
   365→    userId: number;
   366→    operation: string;
   367→    targetType: string;
   368→    targetKey: string;
   369→    idempotencyKeyHash: string;
   370→    requestFingerprint: string | null;
   371→    resultStatus: string; // "applied" | "no_op" | "not_suppressed"
   372→    resultEventId?: string | null;
   373→    resultSuppressionId?: string | null;
   374→  },
   375→): Promise<void> {
   376→  await tx.consentMutationIdempotency.create({
   377→    data: {
   378→      userId: opts.userId,
   379→      operation: opts.operation,
   380→      targetType: opts.targetType,
   381→      targetKey: opts.targetKey,
   382→      idempotencyKeyHash: opts.idempotencyKeyHash,
   383→      requestFingerprint: opts.requestFingerprint,
   384→      resultStatus: opts.resultStatus,
   385→      resultEventId: opts.resultEventId ?? null,
   386→      resultSuppressionId: opts.resultSuppressionId ?? null,
   387→    },
   388→    select: { id: true },
   389→  });
   390→}
   391→
   392→/**
   393→ * Verify a request fingerprint matches the stored one. Throws IdempotencyConflictError
   394→ * if the fingerprints disagree (same key, different mutable payload).
   395→ */
   396→function verifyFingerprint(
   397→  stored: string | null,
   398→  incoming: string | null,
   399→): void {
   400→  if (stored !== null && incoming !== null && stored !== incoming) {
   401→    throw new IdempotencyConflictError(
   402→      "Idempotency key reused with conflicting request payload.",
   403→    );
   404→  }
   405→  // If either is null, we don't enforce (one of the calls didn't supply a payload).
   406→  // The DB row was created on the first call; the retry just returns the stored result.
   407→}

   420→/**
   421→ * Explicitly subscribe a Contact to marketing.
   422→ *
   423→ * Atomic transition (all in one transaction with canonical lock held):
   424→ *   - Acquire pg_advisory_xact_lock(userId, emailKey)
   425→ *   - Re-read Contact state AFTER lock acquisition (no stale snapshot)
   426→ *   - Idempotency check: if (userId, "subscribe", idempotencyKeyHash) exists, verify fingerprint + return idempotent_replay
   427→ *   - If marketingStatus === "subscribed" AND no idempotency key → return no_op (NO fake transition event)
   428→ *   - If marketingStatus === "subscribed" AND idempotency key → append audit row (first time we've seen this key)
   429→ *   - Update Contact: marketingStatus→subscribed, marketingConsentSource→source, marketingConsentAt→now
   430→ *   - Lift any active suppression (Phase 9 only writes unsubscribe/manual — both safe to lift on explicit subscribe)
   431→ *   - Append ContactConsentEvent(subscribed) row
   432→ *   - Append SuppressionEvent(lifted) row IF a suppression was lifted
   433→ *   - Append ContactEvent timeline "contact.subscribed"
   434→ *
   435→ * P2002 from a concurrent insert races us — caught OUTSIDE the tx.
   436→ */
   437→export async function subscribeContact(opts: ConsentOperationOptions): Promise<ConsentResult> {
   438→  const { userId, contactId, source, reason, idempotencyKey, requestId, requestPayload } = opts;
   439→  if (!VALID_CONSENT_SOURCES.has(source)) {
   440→    throw new Error(`Invalid consent source: ${source}`);
   441→  }
   442→
   443→  const operation = CONSENT_OPERATIONS.SUBSCRIBE;
   444→  const idempotencyKeyHash = idempotencyKey ? hashIdempotencyKey(userId, operation, idempotencyKey) : null;
   445→  const requestFingerprint = requestPayload !== undefined ? hashRequestFingerprint(requestPayload) : null;
   446→
   447→  try {
   448→    return await db.$transaction(async (tx) => {
   449→      // 1. Resolve Contact to get email (pre-lock read — for lock acquisition).
   450→      const contact = await tx.contact.findFirst({
   451→        where: { id: contactId, userId },
   452→        select: { id: true, email: true, marketingStatus: true },
   453→      });
   454→      if (!contact) {
   455→        return {
   456→          status: "no_op" as const,
   457→          previousStatus: null,
   458→          newStatus: null,
   459→          eventId: null,
   460→          contactNotFound: true,
   461→        };
   462→      }
   463→
   464→      // 2. Acquire canonical lock for (userId, email).
   465→      await acquireCanonicalLock(tx, userId, contact.email);
   466→
   467→      // 3. Re-read Contact AFTER lock (fresh state — no stale snapshot).
   468→      const fresh = await tx.contact.findFirst({
   469→        where: { id: contactId, userId },
   470→        select: { id: true, email: true, marketingStatus: true },
   471→      });
   472→      if (!fresh) {
   473→        return {
   474→          status: "no_op" as const,
   475→          previousStatus: null,
   476→          newStatus: null,
   477→          eventId: null,
   478→          contactNotFound: true,
   479→        };
   480→      }
   481→      const previousStatus = fresh.marketingStatus as MarketingStatus;
   482→
   483→      // 4. Durable idempotency check (inside tx, after lock — serializable).
   484→      //    Uses the dedicated ConsentMutationIdempotency table so a no_op
   485→      //    outcome can be durably replayed without creating a fake transition.
   486→      const existingIdem = await findExistingIdempotency(
   487→        tx, userId, operation, idempotencyKeyHash,
   488→        TARGET_TYPE_CONTACT, contactTargetKey(contactId),
   489→      );
   490→      if (existingIdem) {
   491→        verifyFingerprint(existingIdem.requestFingerprint, requestFingerprint);
   492→        // Replay the ORIGINAL outcome — even if the contact's state has since
   493→        // changed. The original request was a no_op / applied, and a retry must
   494→        // NOT re-evaluate and potentially apply a different transition.
   495→        if (existingIdem.resultEventId === null) {
   496→          // Original was a no_op — return the current state but mark as replay.
   497→          return {
   498→            status: "idempotent_replay" as const,
   499→            previousStatus,
   500→            newStatus: previousStatus, // unchanged — original was a no_op
   501→            eventId: null,
   502→            contactNotFound: false,
   503→          };
   504→        }
   505→        // Original was an applied transition — return the stored eventId.
   506→        return {
   507→          status: "idempotent_replay" as const,
   508→          previousStatus,
   509→          newStatus: MARKETING_STATUSES.SUBSCRIBED,
   510→          eventId: existingIdem.resultEventId,
   511→          contactNotFound: false,
   512→        };
   513→      }
   514→
   515→      // 5. No-op semantics: already subscribed → NO state mutation, NO fake
   516→      //    transition event, REGARDLESS of whether an idempotency key is present.
   517→      //    The idempotency key (if supplied) is persisted to the durable
   518→      //    idempotency table with resultStatus="no_op" so a retry replays the
   519→      //    no_op rather than re-evaluating against the (possibly changed) state.
   520→      if (previousStatus === MARKETING_STATUSES.SUBSCRIBED) {
   521→        if (idempotencyKeyHash) {
   522→          await persistIdempotencyOutcome(tx, {
   523→            userId, operation,
   524→            targetType: TARGET_TYPE_CONTACT,
   525→            targetKey: contactTargetKey(contactId),
   526→            idempotencyKeyHash,
   527→            requestFingerprint,
   528→            resultStatus: "no_op",
   529→            resultEventId: null,
   530→          });
   531→        }
   532→        return {
   533→          status: "no_op" as const,
   534→          previousStatus,
   535→          newStatus: MARKETING_STATUSES.SUBSCRIBED,
   536→          eventId: null,
   537→          contactNotFound: false,
   538→        };
   539→      }
   540→
   541→      // 6. Apply the mutation (real transition: previousStatus !== subscribed).
   542→      const now = new Date();
   543→      await tx.contact.update({
   544→        where: { id: contact.id },
   545→        data: {
   546→          marketingStatus: MARKETING_STATUSES.SUBSCRIBED,
   547→          marketingConsentSource: source,
   548→          marketingConsentAt: now,
   549→        },
   550→      });
   551→
   552→      // 7. Lift any active suppression as a side effect of explicit subscribe.
   553→      // Phase 9 only writes "unsubscribe" / "manual" suppressions, both safe
   554→      // to lift here. Phase 11 hard_bounce/complaint will need separate logic.
   555→      const currentSuppression = await tx.suppressionEntry.findUnique({
   556→        where: { userId_email: { userId, email: contact.email } },
   557→        select: { id: true, active: true, reason: true, suppressionId: true },
   558→      });
   559→      let liftedSuppressionId: string | null = null;
   560→      if (currentSuppression && currentSuppression.active) {
   561→        await tx.suppressionEntry.update({
   562→          where: { id: currentSuppression.id },
   563→          data: { active: false, liftedAt: now, source },
   564→        });
   565→        liftedSuppressionId = currentSuppression.suppressionId;
   566→        // Append SuppressionEvent("lifted") via createMany(skipDuplicates:true)
   567→        // — conflict-safe without in-tx try/catch.
   568→        await tx.suppressionEvent.createMany({
   569→          data: [{
   570→            userId,
   571→            suppressionId: currentSuppression.id,
   572→            email: contact.email,
   573→            operation: SUPPRESSION_OPERATIONS.UNSUPPRESS,
   574→            action: SUPPRESSION_ACTIONS.LIFTED,
   575→            reason: currentSuppression.reason,
   576→            source,
   577→            idempotencyKeyHash: idempotencyKeyHash ? `${idempotencyKeyHash}:lift` : null,
   578→            requestFingerprint: null, // the lift is a deterministic side effect, not independently idempotent
   579→          }],
   580→          skipDuplicates: true,
   581→        });
   582→      }
   583→
   584→      // 8. Append immutable ConsentEvent. P2002 (if a concurrent insert races
   585→      //    us) propagates out of the tx and is caught by the outer try/catch.
   586→      const consentEvent = await tx.contactConsentEvent.create({
   587→        data: {
   588→          userId,
   589→          contactId: contact.id,
   590→          operation,
   591→          previousStatus,
   592→          newStatus: MARKETING_STATUSES.SUBSCRIBED,
   593→          source,
   594→          reason: reason ?? null,
   595→          idempotencyKeyHash,
   596→          requestFingerprint,
   597→        },
   598→        select: { eventId: true },
   599→      });
   600→
   601→      // 8b. Persist durable idempotency outcome (resultStatus="applied").
   602→      if (idempotencyKeyHash) {
   603→        await persistIdempotencyOutcome(tx, {
   604→          userId, operation,
   605→          targetType: TARGET_TYPE_CONTACT,
   606→          targetKey: contactTargetKey(contactId),
   607→          idempotencyKeyHash,
   608→          requestFingerprint,
   609→          resultStatus: "applied",
   610→          resultEventId: consentEvent.eventId,
   611→          resultSuppressionId: liftedSuppressionId,
   612→        });
   613→      }
   614→
   615→      // 9. Timeline event (idempotent via createMany skipDuplicates when key set).
   616→      const timelineDedupeKey = idempotencyKeyHash
   617→        ? `subscribe:${userId}:${contact.id}:${idempotencyKeyHash}`
   618→        : null;
   619→      await tx.contactEvent.createMany({
   620→        data: [{
   621→          contactId: contact.id,
   622→          type: "contact.subscribed",
   623→          detail: { source, previousStatus, reason: reason ?? null } as any,
   624→          requestId: requestId ?? null,
   625→          dedupeKey: timelineDedupeKey,
   626→        }],
   627→        skipDuplicates: true,
   628→      });
   629→
   630→      return {
   631→        status: "applied" as const,
   632→        previousStatus,
   633→        newStatus: MARKETING_STATUSES.SUBSCRIBED,
   634→        eventId: consentEvent.eventId,
   635→        contactNotFound: false,
   636→      };
   637→    });
   638→  } catch (err: any) {
   639→    if (err instanceof IdempotencyConflictError) throw err;
   640→    if (err?.code === "P2002" && idempotencyKeyHash) {
   641→      // Race: a concurrent insert with the same idempotencyKeyHash won.
   642→      // Fetch the existing idempotency outcome and return idempotent_replay.
   643→      const existingIdem = await db.consentMutationIdempotency.findUnique({
   644→        where: { userId_operation_idempotencyKeyHash: { userId, operation, idempotencyKeyHash } },
   645→        select: {
   646→          requestFingerprint: true, resultStatus: true, resultEventId: true,
   647→          targetType: true, targetKey: true,
   648→        },
   649→      });
   650→      if (existingIdem) {
   651→        if (existingIdem.targetType !== TARGET_TYPE_CONTACT || existingIdem.targetKey !== contactTargetKey(contactId)) {
   652→          throw new IdempotencyConflictError("Idempotency key already used for a different contact target.");
   653→        }
   654→        verifyFingerprint(existingIdem.requestFingerprint, requestFingerprint);
   655→        // Re-read current contact state for the returned previousStatus.
   656→        const fresh = await db.contact.findFirst({
   657→          where: { id: contactId, userId },
   658→          select: { marketingStatus: true },
   659→        });
   660→        const currentStatus = (fresh?.marketingStatus ?? MARKETING_STATUSES.UNKNOWN) as MarketingStatus;
   661→        if (existingIdem.resultEventId === null) {
   662→          return {
   663→            status: "idempotent_replay",
   664→            previousStatus: currentStatus,
   665→            newStatus: currentStatus,
   666→            eventId: null,
   667→            contactNotFound: false,
   668→          };
   669→        }
   670→        return {
   671→          status: "idempotent_replay",
   672→          previousStatus: currentStatus,
   673→          newStatus: MARKETING_STATUSES.SUBSCRIBED,
   674→          eventId: existingIdem.resultEventId,
   675→          contactNotFound: false,
   676→        };
   677→      }
   678→    }
   679→    throw err;
   680→  }
   681→}

   898→          where: { id: contactId, userId },

   919→    }

   921→  }

   924→/**
   925→ * Suppress an email at the tenant level. Optionally also unsubscribe the
   926→ * matching Contact (if contactId is provided and belongs to the tenant).
   927→ *
   928→ * When contactId is provided, routes through unsubscribeContact for the
   929→ * atomic ConsentEvent + SuppressionEvent pair (preserves consent history).
   930→ * When no contactId is provided, performs an email-only suppression with
   931→ * its own audit history.

   933→export async function suppressEmail(opts: SuppressOptions): Promise<SuppressResult> {
   934→  const { userId, email, reason, source, contactId, idempotencyKey, requestId, requestPayload } = opts;
   935→  if (!PHASE9_WRITABLE_REASONS.has(reason)) {
   936→    throw new Error(`Phase 9 cannot write suppression reason: ${reason}`);
   937→  }

   939→    throw new Error(`Invalid suppression source: ${source}`);
   940→  }
   941→  const normalized = normalizeEmail(email);
   942→  if (!isValidEmail(normalized)) {
   943→    throw new Error("Invalid email.");
   944→  }
   945→
   946→  const idempotencyKeyHash = idempotencyKey
   947→    ? hashIdempotencyKey(userId, SUPPRESSION_OPERATIONS.SUPPRESS, idempotencyKey)

   956→      source,

   958→      idempotencyKey,
   959→      requestId,
   960→      requestPayload,
   961→    });

   978→      suppressionId: entry?.suppressionId ?? null,
   979→      email: normalized,
   980→      active: entry?.active ?? false,
   981→      eventId: consent.eventId,
   982→      contactNotFound: false,

   985→
   986→  // Email-only suppression.
   987→  try {
   988→    return await db.$transaction(async (tx) => {
   989→      await acquireCanonicalLock(tx, userId, normalized);

   992→      const existingIdem = await findExistingIdempotency(
   993→        tx, userId, SUPPRESSION_OPERATIONS.SUPPRESS, idempotencyKeyHash,
   994→        TARGET_TYPE_EMAIL, normalized,
   995→      );

   999→          where: { userId_email: { userId, email: normalized } },
  1000→          select: { suppressionId: true, active: true },
  1001→        });

  1003→          status: "idempotent_replay" as const,
  1004→          suppressionId: existingIdem.resultSuppressionId ?? entry?.suppressionId ?? null,
  1005→          email: normalized,
  1006→          active: entry?.active ?? false,
  1007→          eventId: existingIdem.resultEventId,
  1008→          contactNotFound: false,
  1009→        };
  1010→      }
  1011→
  1012→      // No-op: already suppressed with same effective reason → NO state mutation,
  1013→      // NO fake transition event. Idempotency key (if supplied) persisted to
  1014→      // the durable idempotency table with resultStatus="no_op".
  1015→      const currentEntry = await tx.suppressionEntry.findUnique({
  1016→        where: { userId_email: { userId, email: normalized } },
  1017→        select: { id: true, active: true, reason: true, suppressionId: true },
  1018→      });
  1019→      if (currentEntry && currentEntry.active && currentEntry.reason === reason) {
  1020→        if (idempotencyKeyHash) {

  1022→            userId, operation: SUPPRESSION_OPERATIONS.SUPPRESS,
  1023→            targetType: TARGET_TYPE_EMAIL,
  1024→            targetKey: normalized,
  1025→            idempotencyKeyHash,
  1026→            requestFingerprint,
  1027→            resultStatus: "no_op",
  1028→            resultEventId: null,
  1029→            resultSuppressionId: currentEntry.suppressionId,
  1030→          });
  1031→        }
  1032→        return {
  1033→          status: "no_op" as const,
  1034→          suppressionId: currentEntry.suppressionId,
  1035→          email: normalized,
  1036→          active: true,
  1037→          eventId: null,
  1038→          contactNotFound: false,
  1039→        };
  1040→      }
  1041→
  1042→      const entry = await tx.suppressionEntry.upsert({
  1043→        where: { userId_email: { userId, email: normalized } },

  1045→          userId,
  1046→          email: normalized,

  1068→          source,
  1069→          idempotencyKeyHash,
  1070→          requestFingerprint,
  1071→        },
  1072→        select: { eventId: true },
  1073→      });
  1074→
  1075→      // Persist durable idempotency outcome (resultStatus="applied").

  1078→          userId, operation: SUPPRESSION_OPERATIONS.SUPPRESS,
  1079→          targetType: TARGET_TYPE_EMAIL,
  1080→          targetKey: normalized,
  1081→          idempotencyKeyHash,
  1082→          requestFingerprint,
  1083→          resultStatus: "applied",
  1084→          resultEventId: event.eventId,
  1085→          resultSuppressionId: entry.suppressionId,
  1086→        });
  1087→      }
  1088→
  1089→      return {
  1090→        status: "applied" as const,
  1091→        suppressionId: entry.suppressionId,

  1098→  } catch (err: any) {

  1108→      if (existingIdem) {
  1109→        if (existingIdem.targetType !== TARGET_TYPE_EMAIL || existingIdem.targetKey !== normalized) {
  1110→          throw new IdempotencyConflictError("Idempotency key already used for a different email target.");
  1111→        }
  1112→        verifyFingerprint(existingIdem.requestFingerprint, requestFingerprint);
  1113→        const entry = await db.suppressionEntry.findUnique({
  1114→          where: { userId_email: { userId, email: normalized } },
  1115→          select: { suppressionId: true, active: true },
  1116→        });
  1117→        return {
  1118→          status: "idempotent_replay",
  1119→          suppressionId: existingIdem.resultSuppressionId ?? entry?.suppressionId ?? null,
  1120→          email: normalized,
  1121→          active: entry?.active ?? false,
  1122→          eventId: existingIdem.resultEventId,
  1123→          contactNotFound: false,
  1124→        };
  1125→      }
  1126→    }
  1127→    throw err;
  1128→  }
  1129→}
  1130→
  1131→/**
  1132→ * Lift a suppression. By default this ONLY lifts the suppression — it does
  1133→ * NOT subscribe the contact. To resubscribe, pass `alsoSubscribe: true`
  1134→ * (which routes through subscribeContact for the explicit consent action).
  1135→ *
  1136→ * `alsoSubscribe: true` is the ONLY way to combine lift + subscribe. An
  1137→ * ordinary lift call CANNOT silently subscribe.
  1138→ */
  1139→export async function unsuppressEmail(opts: UnsuppressOptions): Promise<UnsuppressResult> {
  1140→  const { userId, email, source, alsoSubscribe, idempotencyKey, requestId, requestPayload } = opts;
  1141→  if (!VALID_CONSENT_SOURCES.has(source)) {
  1142→    throw new Error(`Invalid suppression source: ${source}`);
  1143→  }
  1144→  const normalized = normalizeEmail(email);
  1145→  if (!isValidEmail(normalized)) {
  1146→    throw new Error("Invalid email.");
  1147→  }
  1148→
  1149→  // If alsoSubscribe is set, this is an EXPLICIT consent action — route
  1150→  // through subscribeContact which acquires the canonical lock and performs
  1151→  // the full atomic transition (subscribe + lift suppression).
  1152→  if (alsoSubscribe) {
  1153→    const contact = await db.contact.findUnique({
  1154→      where: { userId_email: { userId, email: normalized } },
  1155→      select: { id: true },
  1156→    });
  1157→    if (!contact) {
  1158→      // No contact — still lift any suppression that might exist.
  1159→      return liftOnly(userId, normalized, source, idempotencyKey, requestId, requestPayload);
  1160→    }
  1161→    const consent = await subscribeContact({
  1162→      userId,
  1163→      contactId: contact.id,
  1164→      source,
  1165→      idempotencyKey,
  1166→      requestId,
  1167→      requestPayload,
  1168→    });
  1169→    return {
  1170→      status: consent.status === "idempotent_replay" ? "idempotent_replay" : "applied",
  1171→      email: normalized,
  1172→      active: false, // subscribeContact lifted it
  1173→      eventId: consent.eventId,
  1174→    };
  1175→  }
  1176→
  1177→  return liftOnly(userId, normalized, source, idempotencyKey, requestId, requestPayload);
  1178→}

  1180→async function liftOnly(
  1181→  userId: number,
  1182→  normalizedEmail: string,
  1183→  source: ConsentSource,
  1184→  idempotencyKey: string | undefined,
  1185→  requestId?: string,
  1186→  requestPayload?: unknown,
  1187→): Promise<UnsuppressResult> {
  1188→  const idempotencyKeyHash = idempotencyKey
  1189→    ? hashIdempotencyKey(userId, SUPPRESSION_OPERATIONS.UNSUPPRESS, idempotencyKey)
  1190→    : null;
  1191→  const requestFingerprint = requestPayload !== undefined ? hashRequestFingerprint(requestPayload) : null;
  1192→
  1193→  try {
  1194→    return await db.$transaction(async (tx) => {
  1195→      await acquireCanonicalLock(tx, userId, normalizedEmail);
  1196→
  1197→      const existingIdem = await findExistingIdempotency(
  1198→        tx, userId, SUPPRESSION_OPERATIONS.UNSUPPRESS, idempotencyKeyHash,
  1199→        TARGET_TYPE_EMAIL, normalizedEmail,
  1200→      );
  1201→      if (existingIdem) {
  1202→        verifyFingerprint(existingIdem.requestFingerprint, requestFingerprint);
  1203→        const entry = await tx.suppressionEntry.findUnique({
  1204→          where: { userId_email: { userId, email: normalizedEmail } },
  1205→          select: { active: true },
  1206→        });
  1207→        // Replay original outcome. If original was not_suppressed (resultEventId null),
  1208→        // return not_suppressed; otherwise return idempotent_replay with the stored eventId.
  1209→        if (existingIdem.resultEventId === null) {
  1210→          return {
  1211→            status: "idempotent_replay" as const,
  1212→            email: normalizedEmail,
  1213→            active: entry?.active ?? false,
  1214→            eventId: null,
  1215→          };
  1216→        }
  1217→        return {
  1218→          status: "idempotent_replay" as const,
  1219→          email: normalizedEmail,
  1220→          active: entry?.active ?? false,
  1221→          eventId: existingIdem.resultEventId,
  1222→        };
  1223→      }
  1224→
  1225→      const entry = await tx.suppressionEntry.findUnique({
  1226→        where: { userId_email: { userId, email: normalizedEmail } },
  1227→        select: { id: true, active: true, reason: true, suppressionId: true },
  1228→      });
  1229→      if (!entry || !entry.active) {
  1230→        // No active suppression — no_op (do NOT create a fake "lift" event).
  1231→        // Persist the no_op outcome to the durable idempotency table so a retry
  1232→        // replays the not_suppressed result rather than re-evaluating.
  1233→        if (idempotencyKeyHash) {
  1234→          await persistIdempotencyOutcome(tx, {
  1235→            userId, operation: SUPPRESSION_OPERATIONS.UNSUPPRESS,
  1236→            targetType: TARGET_TYPE_EMAIL,
  1237→            targetKey: normalizedEmail,
  1238→            idempotencyKeyHash,
  1239→            requestFingerprint,
  1240→            resultStatus: "not_suppressed",
  1241→            resultEventId: null,
  1242→            resultSuppressionId: entry?.suppressionId ?? null,
  1243→          });
  1244→        }
  1245→        return {
  1246→          status: "not_suppressed" as const,
  1247→          email: normalizedEmail,
  1248→          active: false,
  1249→          eventId: null,
  1250→        };
  1251→      }
  1252→
  1253→      const now = new Date();
  1254→      await tx.suppressionEntry.update({
  1255→        where: { id: entry.id },
  1256→        data: { active: false, liftedAt: now, source },
  1257→      });
  1258→
  1259→      const event = await tx.suppressionEvent.create({
  1260→        data: {
  1261→          userId,
  1262→          suppressionId: entry.id,
  1263→          email: normalizedEmail,
  1264→          operation: SUPPRESSION_OPERATIONS.UNSUPPRESS,
  1265→          action: SUPPRESSION_ACTIONS.LIFTED,
  1266→          reason: entry.reason,
  1267→          source,
  1268→          idempotencyKeyHash,
  1269→          requestFingerprint,
  1270→        },
  1271→        select: { eventId: true },
  1272→      });
  1273→
  1274→      // Persist durable idempotency outcome (resultStatus="applied").
  1275→      if (idempotencyKeyHash) {
  1276→        await persistIdempotencyOutcome(tx, {
  1277→          userId, operation: SUPPRESSION_OPERATIONS.UNSUPPRESS,
  1278→          targetType: TARGET_TYPE_EMAIL,
  1279→          targetKey: normalizedEmail,
  1280→          idempotencyKeyHash,
  1281→          requestFingerprint,
  1282→          resultStatus: "applied",
  1283→          resultEventId: event.eventId,
  1284→          resultSuppressionId: entry.suppressionId,
  1285→        });
  1286→      }
  1287→
  1288→      // Optional timeline event for the matching contact.
  1289→      const contact = await tx.contact.findUnique({
  1290→        where: { userId_email: { userId, email: normalizedEmail } },
  1291→        select: { id: true },
  1292→      });
  1293→      if (contact) {
  1294→        await tx.contactEvent.createMany({
  1295→          data: [{
  1296→            contactId: contact.id,
  1297→            type: "contact.unsuppressed",
  1298→            detail: { source, reason: entry.reason } as any,
  1299→            requestId: requestId ?? null,
  1300→            dedupeKey: idempotencyKeyHash
  1301→              ? `unsuppress:${userId}:${contact.id}:${idempotencyKeyHash}`
  1302→              : null,
  1303→          }],
  1304→          skipDuplicates: true,
  1305→        });
  1306→      }
  1307→
  1308→      return {
  1309→        status: "applied" as const,
  1310→        email: normalizedEmail,
  1311→        active: false,
  1312→        eventId: event.eventId,
  1313→      };
  1314→    });
  1315→  } catch (err: any) {
  1316→    if (err instanceof IdempotencyConflictError) throw err;
  1317→    if (err?.code === "P2002" && idempotencyKeyHash) {
  1318→      const existingIdem = await db.consentMutationIdempotency.findUnique({
  1319→        where: { userId_operation_idempotencyKeyHash: { userId, operation: SUPPRESSION_OPERATIONS.UNSUPPRESS, idempotencyKeyHash } },
  1320→        select: {
  1321→          requestFingerprint: true, resultStatus: true, resultEventId: true,
  1322→          targetType: true, targetKey: true,
  1323→        },
  1324→      });
  1325→      if (existingIdem) {
  1326→        if (existingIdem.targetType !== TARGET_TYPE_EMAIL || existingIdem.targetKey !== normalizedEmail) {
  1327→          throw new IdempotencyConflictError("Idempotency key already used for a different email target.");
  1328→        }
  1329→        verifyFingerprint(existingIdem.requestFingerprint, requestFingerprint);
  1330→        return {
  1331→          status: "idempotent_replay",
  1332→          email: normalizedEmail,
  1333→          active: false,
  1334→          eventId: existingIdem.resultEventId,
  1335→        };
  1336→      }
  1337→    }
  1338→    throw err;
  1339→  }
  1340→}

  1640→/**
  1641→ * Verify the consent history transition chain for a contact is coherent:
  1642→ * for events ordered by createdAt asc, event[n].previousStatus === event[n-1].newStatus.
  1643→ * Used by tests to assert the serializability invariant.
  1644→ */
  1645→export async function verifyConsentHistoryChain(
  1646→  userId: number,
  1647→  contactId: number,
  1648→): Promise<{ coherent: boolean; events: ConsentHistoryEntry[] }> {
  1649→  const contact = await db.contact.findFirst({
  1650→    where: { id: contactId, userId },
  1651→    select: { id: true, marketingStatus: true },
  1652→  });
  1653→  if (!contact) return { coherent: false, events: [] };
  1654→
  1655→  const rows = await db.contactConsentEvent.findMany({
  1656→    where: { userId, contactId },
  1657→    orderBy: { createdAt: "asc" },
  1658→  });
  1659→
  1660→  let coherent = true;
  1661→  for (let i = 1; i < rows.length; i++) {
  1662→    if (rows[i].previousStatus !== rows[i - 1].newStatus) {
  1663→      coherent = false;
  1664→      break;
  1665→    }
  1666→  }
  1667→  // Final Contact state must equal last event's newStatus.
  1668→  if (rows.length > 0 && rows[rows.length - 1].newStatus !== contact.marketingStatus) {
  1669→    coherent = false;
  1670→  }
  1671→
  1672→  return {
  1673→    coherent,
  1674→    events: rows.map((r) => ({
  1675→      eventId: r.eventId,
  1676→      contactId: r.contactId,
  1677→      operation: r.operation,
  1678→      previousStatus: r.previousStatus,
  1679→      newStatus: r.newStatus,
  1680→      source: r.source,
